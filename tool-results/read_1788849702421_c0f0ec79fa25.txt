   246→  const handleDragEnd = useCallback(
   247→    async (event: DragEndEvent) => {

   249→      if (!over || active.id === over.id) return;

   251→      const newIndex = activeHabits.findIndex((h) => h.id === over.id);

   253→

   255→      // Re-assign `order` so the new array position matches the DB order

   257→      const updates: { id: string; order: number }[] = [];
   258→      reordered.forEach((h, idx) => {
   259→        if (h.order !== idx) updates.push({ id: h.id, order: idx });
   260→      });

   262→      // Optimistic local override: reordered active habits (with new order
   263→      // field) followed by the unchanged paused/archived habits.
   264→      const activeIds = new Set(reordered.map((h) => h.id));

   267→        ...reordered.map((h, idx) => ({ ...h, order: idx })),

   278→            fetch(`/api/habits/${u.id}`, {
   279→              method: 'PUT',
   280→              headers: { 'Content-Type': 'application/json' },

   283→          ),

   291→        toast.error('Gagal menyimpan urutan');
   292→        // Revert to server truth.
   293→        await queryClient.invalidateQueries({ queryKey: ['habits'] });
   294→        setLocalHabitsOverride(null);
   295→      }
   296→    },
   297→    [activeHabits, habits, queryClient],
   298→  );

   361→  // BUGHUNT-ROUND2 RACE-1: `isCancelled` guard (passed by the loading

   396→      setCompletedAtMap(atMap);

   413→    }

   415→    // RACE-1: a newer fetch may have resolved while this one was in flight.
   416→    // Bailing here avoids: (a) painting stale completion maps over the
   417→    // current date's state, (b) flipping cachedMonthRef back to the stale
   418→    // month (which would only cost a redundant re-fetch later, but still).
   419→    if (isCancelled?.()) return;
   420→
   421→    const monthCache: Record<string, HabitLog[]> = {};
   422→    const map: Record<string, boolean> = {};
   423→    const atMap: Record<string, string> = {};
   424→
   425→    active.forEach((habit) => {
   426→      const logs = groupedLogs[habit.id] || [];
   427→      monthCache[habit.id] = logs;
   428→      const dayLog = logs.find((l) => toDateString(l.date) === date);
   429→      map[habit.id] = dayLog?.completed ?? false;
   430→      if (dayLog?.completedAt) {
   431→        atMap[habit.id] = formatJakartaTime(dayLog.completedAt);
   432→      }
   433→    });
   434→
   435→    monthLogsCacheRef.current[month] = monthCache;
   436→    cachedMonthRef.current = month;
   437→    // BUG-16 fix: record the refreshKey that populated this cache so a future
   438→    // refreshKey change invalidates the cache.
   439→    cachedRefreshKeyRef.current = refreshKey;
   440→    setCompletionMap(map);
   441→    setCompletedAtMap(atMap);
   442→  };

   454→        pendingSaveRef.current = null;
   455→        if (!pending) return;
   456→        try {
   457→          await fetch('/api/daily-logs', {
   458→            method: 'POST',
   459→            headers: { 'Content-Type': 'application/json' },
   460→            body: JSON.stringify(pending),
   461→          });
   462→          queryClient.invalidateQueries({ queryKey: ['daily-logs', pending.date] });
   463→        } catch {
   464→          toast.error('Gagal menyimpan catatan');
   465→        }
   466→      }, 600);
   467→    },
   468→    [selectedDate, queryClient],
   469→  );

   472→    (html: string) => {
   473→      setNotes(html);
   474→      debouncedSave({ notes: html });
   475→    },
   476→    [debouncedSave],

   479→  // PHASE4-POLISH: visible-text length for the "X karakter" hint. The stored

   481→  // tags etc. — which would be misleading. Strip tags to get the user-visible
   482→  // length. Returns 0 for empty/whitespace-only content.
   483→  const notesCharCount = useMemo(() => {
   484→    if (!notes) return 0;
   485→    // Quick + dirty tag stripper — sufficient for the count display only.
   486→    // (For rendering, the TipTap editor handles its own HTML safely.)
   487→    return notes.replace(/<[^>]+>/g, '').replace(/&nbsp;/g, ' ').trim().length;
   488→  }, [notes]);

   490→  // ---- handlers ----
   491→  // Ref to track the element that triggered a habit completion (for confetti position).
   492→  // Set in handleHabitCheck, read in toggleHabit after successful API response.
   493→  const confettiElRef = useRef<HTMLElement | null>(null);
   494→
   495→  // PERF-REACT-1 fix: toggleHabit is declared BEFORE handleHabitCheck and
   496→  // handleTimeDialogSubmit (which call it) so the useCallback deps arrays
   497→  // can reference it without temporal-dead-zone errors. Reads
   498→  // `completionMap` via `completionMapRef.current` (not directly) so the
   499→  // callback identity stays stable across toggles — this is what lets
