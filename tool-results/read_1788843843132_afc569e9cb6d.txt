   412→   412→//    pointer, dimmed backdrop (tap/Escape to close).
   413→   413→//  • Mobile-only (md:hidden). a11y: aria-current, aria-expanded,
   414→   414→//    role=menu, focus-visible rings, prefers-reduced-motion respected.
   415→   415→
   416→   416→const DOCK_H = 62; // dock height (px)
   417→   417→const DOCK_SIDE = 0.375; // width share of each tab group (left/right)
   418→   418→const IND_INSET = 3; // indicator horizontal inset inside a tab
   419→   419→const FAB_SIZE = 56; // FAB diameter (px)
   420→   420→const FAB_PROTRUDE = 22; // px of FAB protruding above the dock top edge
   421→   421→
   422→   422→const NAV_LEFT_ITEMS: { id: TabId; label: string; icon: React.ElementType }[] = [
   423→   423→  { id: 'dashboard', label: 'Beranda', icon: LayoutDashboard },
   424→   424→  { id: 'tracker', label: 'Track', icon: CheckSquare },
   425→   425→];
   426→   426→
   427→   427→const NAV_RIGHT_ITEMS: { id: TabId; label: string; icon: React.ElementType }[] = [
   428→   428→  { id: 'finance', label: 'Keuangan', icon: Wallet },
   429→   429→  { id: 'settings', label: 'Pengaturan', icon: SettingsIcon },
   430→   430→];
   431→   431→
   432→   432→// Left-edge x-offset of each dock tab, as a fraction of dock width.
   433→   433→// `goals` is NOT in the dock (sidebar-only) → undefined → indicator hides.
   434→   434→const TAB_X_FRACTION: Partial<Record<TabId, number>> = {
   435→   435→  dashboard: 0,
   436→   436→  tracker: DOCK_SIDE / 2,
   437→   437→  finance: 1 - DOCK_SIDE,
   438→   438→  settings: 1 - DOCK_SIDE / 2,
   439→   439→};
   440→   440→
   441→   441→function PremiumBottomNav({
   442→   442→  activeTab,
   443→   443→  onNavClick,
   444→   444→}: {
   445→   445→  activeTab: TabId;
   446→   446→  onNavClick: (id: TabId) => void;
   447→   447→}) {
   448→   448→  const dockRef = useRef<HTMLDivElement>(null);
   449→   449→  const [dockW, setDockW] = useState(0);
   450→   450→  const [fabOpen, setFabOpen] = useState(false);
   451→   451→  // BUGHUNT-ROUND2 FAB-1: quick-add trigger lives in the store so this
   452→   452→  // deep nav component can fire it without prop-drilling from Home().
   453→   453→  const triggerQuickAdd = useAppStore((s) => s.triggerQuickAdd);
   454→   454→
   455→   455→  useLayoutEffect(() => {
   456→   456→    if (!dockRef.current) return;
   457→   457→    const update = () => setDockW(dockRef.current?.offsetWidth ?? 0);
   458→   458→    update();
   459→   459→    const observer = new ResizeObserver(update);
   460→   460→    observer.observe(dockRef.current);
   461→   461→    return () => observer.disconnect();
   462→   462→  }, []);
   463→   463→
   464→   464→  // Close popup on Escape (a11y parity with the tap-to-close backdrop).
   465→   465→  useEffect(() => {
   466→   466→    if (!fabOpen) return;
   467→   467→    const onKey = (e: KeyboardEvent) => {
   468→   468→      if (e.key === 'Escape') setFabOpen(false);
   469→   469→    };
   470→   470→    window.addEventListener('keydown', onKey);
   471→   471→    return () => window.removeEventListener('keydown', onKey);
   472→   472→  }, [fabOpen]);
   473→   473→
   474→   474→  // Liquid indicator geometry. GPU-friendly: fixed width per tab +
   475→   475→  // translateX transition with a spring (overshoot) easing curve.
   476→   476→  const tabW = (dockW * DOCK_SIDE) / 2;
   477→   477→  const activeFraction = TAB_X_FRACTION[activeTab];
   478→   478→  const indVisible = activeFraction !== undefined && dockW > 0;
   479→   479→  const indX = indVisible ? activeFraction * dockW + IND_INSET : 0;
   480→   480→
   481→   481→  const renderTab = (item: { id: TabId; label: string; icon: React.ElementType }) => {
   482→   482→    const Icon = item.icon;
   483→   483→    const isActive = activeTab === item.id;
   484→   484→    return (
   485→   485→      <button
   486→   486→        key={item.id}

   490→   490→        className={cn(
   491→   491→          'relative z-10 flex flex-1 flex-col items-center justify-center gap-[3px]',
   492→   492→          'whitespace-nowrap',
   493→   493→          'transition-transform duration-150 motion-reduce:transition-none active:scale-[0.94]',
   494→   494→          'focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-teal-400/70',
   495→   495→          'focus-visible:ring-offset-1 focus-visible:rounded-2xl'
   496→   496→        )}
   497→   497→      >
   498→   498→        <span
   499→   499→          aria-hidden="true"
   500→   500→          className={cn('grid place-items-center', isActive && 'anim-nav-icon-pop')}
   501→   501→        >
   502→   502→          <Icon
   503→   503→            className={cn(
   504→   504→              'h-[21px] w-[21px] shrink-0 transition-colors duration-300 motion-reduce:transition-none',
   505→   505→              isActive
   506→   506→                ? 'text-white drop-shadow-[0_1px_1px_rgba(0,66,55,0.25)]'
   507→   507→                : 'text-slate-500 dark:text-slate-400'
   508→   508→            )}
   509→   509→            strokeWidth={isActive ? 2.4 : 1.7}
   510→   510→          />
   511→   511→        </span>
   512→   512→        <span
   513→   513→          className={cn(
   514→   514→            'premium-dock-label transition-colors duration-300 motion-reduce:transition-none',
   515→   515→            isActive
   516→   516→              ? 'font-bold text-white'
   517→   517→              : 'font-medium text-slate-500 dark:text-slate-400'
   518→   518→          )}
   519→   519→        >
   520→   520→          {item.label}
   521→   521→        </span>
   522→   522→      </button>
   523→   523→    );
   524→   524→  };
   525→   525→
   526→   526→  return (
   527→   527→    <div className="fixed bottom-0 inset-x-0 z-40 md:hidden">
   528→   528→      {/* Dimmed backdrop while the quick-add popup is open — tap to close.
   529→   529→          Rendered OUTSIDE the animated nav element (no transformed
   530→   530→          ancestor) so position:fixed is always viewport-correct. */}
   531→   531→      {fabOpen && (
   532→   532→        <div
   533→   533→          className="fixed inset-0 z-30 bg-slate-950/25 dark:bg-black/40 backdrop-blur-[2px] anim-fab-backdrop"
   534→   534→          onClick={() => setFabOpen(false)}
   535→   535→          aria-hidden="true"
   536→   536→        />
   537→   537→      )}
   538→   538→
   539→   539→      <nav
   540→   540→        ref={dockRef}
   541→   541→        aria-label="Navigasi utama"
   542→   542→        className={cn(
   543→   543→          'relative z-40 mx-3 mb-[calc(env(safe-area-inset-bottom)+10px)]',
   544→   544→          'anim-nav-dock-enter',
   545→   545→          'rounded-[26px]',
   546→   546→          'bg-white/80 dark:bg-slate-900/80',
   547→   547→          'backdrop-blur-2xl backdrop-saturate-150',
   548→   548→          'border border-slate-900/[0.07] dark:border-white/10',
   549→   549→          'premium-dock-shadow'
   550→   550→        )}
   551→   551→        style={{ height: DOCK_H }}
   552→   552→      >
   553→   553→        {/* Glass hairline highlight along the top edge (light refraction) */}
   554→   554→        <div
   555→   555→          aria-hidden="true"
   556→   556→          className="absolute inset-x-[16px] top-0 h-px rounded-full bg-gradient-to-r from-transparent via-white/80 to-transparent dark:via-white/15 pointer-events-none"
   557→   557→        />
   558→   558→
   559→   559→        {/* Liquid gradient indicator — slides between tabs, passes behind
   560→   560→            the FAB (z-0 vs z-20). Fades out when a non-dock tab (goals)
   561→   561→            is active. */}
   562→   562→        <div
   563→   563→          aria-hidden="true"
   564→   564→          className="nav-liquid-indicator absolute left-0 top-[7px] z-0 rounded-full"
   565→   565→          style={{
   566→   566→            height: DOCK_H - 14,
   567→   567→            width: Math.max(0, tabW - IND_INSET * 2),
   568→   568→            transform: `translateX(${indX}px)`,
   569→   569→            opacity: indVisible ? 1 : 0,
   570→   570→          }}
   571→   571→        />
   572→   572→
   573→   573→        {/* Soft teal halo behind the FAB — glows through the glass dock */}
   574→   574→        <div

   577→   577→        />

   595→   595→            style={{ bottom: DOCK_H + FAB_PROTRUDE + 10 }}

   610→   610→              // BUGHUNT-ROUND2 FAB-1: was just onNavClick('finance') — the
