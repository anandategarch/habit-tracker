   402→// ── PremiumBottomNav ───────────────────────────────────────────────────
   403→// PREMIUM REDESIGN (replaces the old Flutter notched bar):
   404→// "Floating Glass Dock" — iOS-18 / modern-fintech design language.
   405→//  • Floating frosted-glass bar (mx-3, rounded-[26px], backdrop-blur-2xl)
   406→//    with layered teal-tinted depth shadows + top hairline highlight.
   407→//  • Gradient "liquid" indicator that slides between tabs with a spring
   408→//    curve (cubic-bezier overshoot) and passes BEHIND the FAB.
   409→//  • Center FAB: teal→emerald gradient, glow, socket ring illusion,
   410→//    rotates 45° into an X when the menu is open.
   411→//  • Quick-add popup: frosted glass card, spring-staggered items, tail
   412→//    pointer, dimmed backdrop (tap/Escape to close).
   413→//  • Mobile-only (md:hidden). a11y: aria-current, aria-expanded,
   414→//    role=menu, focus-visible rings, prefers-reduced-motion respected.

   416→const DOCK_H = 62; // dock height (px)
   417→const DOCK_SIDE = 0.375; // width share of each tab group (left/right)
   418→const IND_INSET = 3; // indicator horizontal inset inside a tab
   419→const FAB_SIZE = 56; // FAB diameter (px)
   420→const FAB_PROTRUDE = 22; // px of FAB protruding above the dock top edge
   421→
   422→const NAV_LEFT_ITEMS: { id: TabId; label: string; icon: React.ElementType }[] = [
   423→  { id: 'dashboard', label: 'Beranda', icon: LayoutDashboard },
   424→  { id: 'tracker', label: 'Track', icon: CheckSquare },
   425→];

   427→const NAV_RIGHT_ITEMS: { id: TabId; label: string; icon: React.ElementType }[] = [
   428→  { id: 'finance', label: 'Keuangan', icon: Wallet },
   429→  { id: 'settings', label: 'Pengaturan', icon: SettingsIcon },
   430→];
   431→
   432→// Left-edge x-offset of each dock tab, as a fraction of dock width.
   433→// `goals` is NOT in the dock (sidebar-only) → undefined → indicator hides.
   434→const TAB_X_FRACTION: Partial<Record<TabId, number>> = {
   435→  dashboard: 0,
   436→  tracker: DOCK_SIDE / 2,
   437→  finance: 1 - DOCK_SIDE,
   438→  settings: 1 - DOCK_SIDE / 2,
   439→};

   441→function PremiumBottomNav({
   442→  activeTab,
   443→  onNavClick,
   444→}: {
   445→  activeTab: TabId;
   446→  onNavClick: (id: TabId) => void;
   447→}) {
   448→  const dockRef = useRef<HTMLDivElement>(null);
   449→  const [dockW, setDockW] = useState(0);
   450→  const [fabOpen, setFabOpen] = useState(false);
   451→  // BUGHUNT-ROUND2 FAB-1: quick-add trigger lives in the store so this
   452→  // deep nav component can fire it without prop-drilling from Home().
   453→  const triggerQuickAdd = useAppStore((s) => s.triggerQuickAdd);
   454→
   455→  useLayoutEffect(() => {
   456→    if (!dockRef.current) return;
   457→    const update = () => setDockW(dockRef.current?.offsetWidth ?? 0);
   458→    update();
   459→    const observer = new ResizeObserver(update);
   460→    observer.observe(dockRef.current);
   461→    return () => observer.disconnect();
   462→  }, []);
   463→
   464→  // Close popup on Escape (a11y parity with the tap-to-close backdrop).
   465→  useEffect(() => {
   466→    if (!fabOpen) return;
   467→    const onKey = (e: KeyboardEvent) => {
   468→      if (e.key === 'Escape') setFabOpen(false);
   469→    };
   470→    window.addEventListener('keydown', onKey);
   471→    return () => window.removeEventListener('keydown', onKey);
   472→  }, [fabOpen]);
   473→
   474→  // Liquid indicator geometry. GPU-friendly: fixed width per tab +
   475→  // translateX transition with a spring (overshoot) easing curve.
   476→  const tabW = (dockW * DOCK_SIDE) / 2;
   477→  const activeFraction = TAB_X_FRACTION[activeTab];
   478→  const indVisible = activeFraction !== undefined && dockW > 0;
   479→  const indX = indVisible ? activeFraction * dockW + IND_INSET : 0;
   480→
   481→  const renderTab = (item: { id: TabId; label: string; icon: React.ElementType }) => {
   482→    const Icon = item.icon;
   483→    const isActive = activeTab === item.id;
