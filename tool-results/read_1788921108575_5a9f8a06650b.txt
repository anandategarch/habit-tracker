   293→ const handleDragEnd = useCallback(
   294→   async (event: DragEndEvent) => {
   295→     const { active, over } = event;
   296→     if (!over || active.id === over.id) return;
   297→     const oldIndex = activeHabits.findIndex((h) => h.id === active.id);
   298→     const newIndex = activeHabits.findIndex((h) => h.id === over.id);
   299→     if (oldIndex < 0 || newIndex < 0) return;
   300→
   301→     const reordered = arrayMove(activeHabits, oldIndex, newIndex);
   302→     // Re-assign `order` so the new array position matches the DB order
   303→     // (0..N-1 across active habits). Collect only the diffs to PUT.
   304→     const updates: { id: string; order: number }[] = [];
   305→     reordered.forEach((h, idx) => {
   306→       if (h.order !== idx) updates.push({ id: h.id, order: idx });
   307→     });
   308→
   309→     // Optimistic local override: reordered active habits (with new order
   310→     // field) followed by the unchanged paused/archived habits.
   311→     const activeIds = new Set(reordered.map((h) => h.id));
   312→     const nonActive = habits.filter((h) => !activeIds.has(h.id));
   313→     const reorderedAll: Habit[] = [
   314→       ...reordered.map((h, idx) => ({ ...h, order: idx })),
   315→       ...nonActive,
   316→     ];
   317→     setLocalHabitsOverride(reorderedAll);
   318→
   319→     // Persist each changed habit's order via PUT /api/habits/[id].
   320→     // Fire-and-forget in parallel; invalidate the query on settle so the
   321→     // server-side truth is re-fetched (and the local override cleared).
   322→     try {
   323→       await Promise.all(
   324→         updates.map((u) =>
   325→           fetch(`/api/habits/${u.id}`, {
   326→             method: 'PUT',
   327→             headers: { 'Content-Type': 'application/json' },
   328→             body: JSON.stringify({ order: u.order }),
   329→           }),
   330→         ),
   331→       );
   332→       await queryClient.invalidateQueries({ queryKey: ['habits'] });
   333→       // Brief delay so the refetch lands before we drop the override —
   334→       // otherwise a re-render with the stale query cache could flicker
   335→       // back to the old order for one frame.
   336→       setTimeout(() => setLocalHabitsOverride(null), 200);
   337→     } catch {
   338→       toast.error('Gagal menyimpan urutan');
   339→       // Revert to server truth.
   340→       await queryClient.invalidateQueries({ queryKey: ['habits'] });
   341→       setLocalHabitsOverride(null);
   342→     }
   343→   },
   344→   [activeHabits, habits, queryClient],
   345→ );
   346→
   347→ /** Toggle drag mode. When enabling, force viewFilter to "all" so every

   385→ // Best current streak across all active habits

   388→   const cache = monthLogsCacheRef.current[month];
   389→   if (!cache) return 0;
   390→   let best = 0;
   391→   for (const h of activeHabits) {
   392→     const logs = cache[h.id] || [];
   393→     // PHASE1-HABIT: pass vacationMode so vacationing habits' streaks don't

   400→       startDate: h.startDate,

   407→ // ---- fetch completions (month-cached) ----
   408→ // BUGHUNT-ROUND2 RACE-1: `isCancelled` guard (passed by the loading
   409→ // effect) prevents a SLOWER stale fetch (e.g. for the previous date/
   410→ // month after rapid navigation) from overwriting the state of the
   411→ // NEWER fetch that already resolved. Previously only `setLoading` was
   412→ // guarded — the completion maps could be painted with the wrong day's
   413→ // data.
   414→ const fetchCompletions = async (
   415→   habitList: Habit[],
   416→   date: string,
   417→   isCancelled?: () => boolean,
   418→ ) => {
   419→   const month = date.slice(0, 7);
   420→
   421→   // BUG-16 fix: include refreshKey in the cache hit check. Without this,
   422→   // changing refreshKey (e.g. after creating a habit) would still hit the
   423→   // stale cache and never re-fetch.
   424→   if (
   425→     cachedMonthRef.current === month &&
   426→     monthLogsCacheRef.current[month] &&
   427→     cachedRefreshKeyRef.current === refreshKey
   428→   ) {
   429→     const cache = monthLogsCacheRef.current[month];
   430→     const map: Record<string, boolean> = {};
   431→     const atMap: Record<string, string> = {};
   432→     habitList
   433→       .filter((h) => h.status === 'active')
   434→       .forEach((h) => {
   435→         const logs = cache[h.id] || [];
   436→         const dayLog = logs.find((l) => toDateString(l.date) === date);
   437→         map[h.id] = dayLog?.completed ?? false;
   438→         if (dayLog?.completedAt) {
   439→           atMap[h.id] = formatJakartaTime(dayLog.completedAt);
   440→         }
   441→       });
   442→     setCompletionMap(map);
   443→     setCompletedAtMap(atMap);
   444→     return;
   445→   }
   446→
   447→   const active = habitList.filter((h) => h.status === 'active');
   448→   const ids = active.map((h) => h.id);
   449→
   450→   let groupedLogs: Record<string, HabitLog[]> = {};
   451→   try {
   452→     const res = await fetch(
   453→       `/api/habits/batch-logs?month=${month}&habitIds=${ids.join(',')}`,
   454→     );
   455→     if (res.ok) {
   456→       groupedLogs = await res.json();
   457→     }
   458→   } catch {
   459→     // fall through to empty defaults
   460→   }
   461→
   462→   // RACE-1: a newer fetch may have resolved while this one was in flight.
   463→   // Bailing here avoids: (a) painting stale completion maps over the
   464→   // current date's state, (b) flipping cachedMonthRef back to the stale
   465→   // month (which would only cost a redundant re-fetch later, but still).
   466→   if (isCancelled?.()) return;
   467→
   468→   const monthCache: Record<string, HabitLog[]> = {};
   469→   const map: Record<string, boolean> = {};
   470→   const atMap: Record<string, string> = {};
   471→
   472→   active.forEach((habit) => {
   473→     const logs = groupedLogs[habit.id] || [];
   474→     monthCache[habit.id] = logs;
   475→     const dayLog = logs.find((l) => toDateString(l.date) === date);
   476→     map[habit.id] = dayLog?.completed ?? false;
   477→     if (dayLog?.completedAt) {
   478→       atMap[habit.id] = formatJakartaTime(dayLog.completedAt);
   479→     }
   480→   });
   481→
   482→   monthLogsCacheRef.current[month] = monthCache;
   483→   cachedMonthRef.current = month;
   484→   // BUG-16 fix: record the refreshKey that populated this cache so a future
   485→   // refreshKey change invalidates the cache.
   486→   cachedRefreshKeyRef.current = refreshKey;
   487→   setCompletionMap(map);
   488→   setCompletedAtMap(atMap);
   489→ };

   491→ // ---- debounced save (notes only) ----
   492→ const debouncedSave = useCallback(
   493→   (patch: { notes?: string }) => {
   494→     if (saveTimerRef.current) clearTimeout(saveTimerRef.current);
   495→     // BUG-19 fix: stash the pending patch (with the current selectedDate)
