    58→// ── MiniProgressRing: circular progress ring for histogram ──────────────
    59→
    60→function MiniProgressRing({
    61→  percentage,
    62→  size = 48,
    63→  strokeWidth = 3.5,
    64→  color,
    65→  isHighlighted = false,
    66→  children,
    67→}: {
    68→  percentage: number;
    69→  size?: number;
    70→  strokeWidth?: number;
    71→  color: string;
    72→  isHighlighted?: boolean;
    73→  children?: React.ReactNode;
    74→}) {
    75→  const radius = (size - strokeWidth) / 2;
    76→  const circumference = 2 * Math.PI * radius;
    77→  const clampedPct = Math.min(Math.max(percentage, 0), 100);
    78→  const offset = circumference - (clampedPct / 100) * circumference;
    79→
    80→  return (
    81→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
    82→      <svg width={size} height={size} className="-rotate-90">
    83→        {/* Highlight glow ring for dominant bucket */}
    84→        {isHighlighted && (
    85→          <circle
    86→            cx={size / 2}
    87→            cy={size / 2}
    88→            r={radius + 1.5}
    89→            fill="none"
    90→            stroke={color}
    91→            strokeWidth="1"
    92→            opacity="0.2"
    93→          />
    94→        )}
    95→        {/* Background track */}
    96→        <circle
    97→          cx={size / 2}
    98→          cy={size / 2}
    99→          r={radius}
   100→          fill="none"
   101→          stroke="currentColor"
   102→          strokeWidth={strokeWidth}
   103→          className="text-muted/20"
   104→        />
   105→        {/* Progress arc */}
   106→        <circle
   107→          cx={size / 2}
   108→          cy={size / 2}
   109→          r={radius}
   110→          fill="none"
   111→          stroke={color}
   112→          strokeWidth={strokeWidth}
   113→          strokeDasharray={circumference}
   114→          strokeDashoffset={offset}
   115→          strokeLinecap="round"
   116→          className="transition-all duration-700 ease-out"
   117→          opacity={isHighlighted ? 1 : 0.5}
   118→        />
   119→      </svg>
   120→      <div className="absolute inset-0 flex items-center justify-center">
   121→        {children}
   122→      </div>
   123→    </div>
   124→  );
   125→}

   127→// ── Module-scope constants (hoisted out of component for reuse) ──────────
   128→//
   129→// PERF-REACT-1 fix: `new Intl.DateTimeFormat(...)` was being constructed
   130→// inside per-transaction loops (peak-hour + time-of-day), creating hundreds
   131→// of formatter objects per render for a category with many transactions.
   132→// Hoisting the formatter to module scope means it is created once on module
   133→// load and reused for the lifetime of the app.
   134→const JAKARTA_HOUR_FORMAT = new Intl.DateTimeFormat('en-GB', {
   135→  timeZone: 'Asia/Jakarta',
   136→  hour: '2-digit',
   137→  hour12: false,
   138→});

   140→const DOW_NAMES = ['Min', 'Sen', 'Sel', 'Rab', 'Kam', 'Jum', 'Sab'];

   142→// ── CategoryDetailView ─────────────────────────────────────────────────
   143→
   144→export interface CategoryDetailViewProps {
   145→  cat: CategoryTotal;
   146→  transactions: Transaction[];
   147→  prevTransactions: Transaction[];
   148→  selectedMonth: string;
   149→  monthOptions: { value: string; label: string }[];
   150→  onBack: () => void;
   151→  onSelectMonth: (m: string) => void;
   152→}
   153→
   154→export function CategoryDetailView({
   155→  cat,
   156→  transactions,
   157→  prevTransactions,
   158→  selectedMonth,
   159→  monthOptions,
   160→  onBack,
   161→  onSelectMonth,
   162→}: CategoryDetailViewProps) {
   163→  // ONE-CLICK-8 (Task 4-b A.7): 1-tap jump from a category's analytics
   164→  // detail to the Transactions sub-tab with this category's filter applied
   165→  // (openFinanceFocus). The explorer's month is mirrored to the global month
   166→  // picker first so the filtered list shows the same period being explored.
   167→  const setStoreMonth = useAppStore(s => s.setSelectedMonth);
   168→  const openFinanceFocus = useAppStore(s => s.openFinanceFocus);
   169→
   170→  const handleViewTransactions = () => {
   171→    setStoreMonth(selectedMonth);
   172→    openFinanceFocus({ category: cat.name });
   173→  };
   174→
   175→  // PERF-REACT-1 fix: ALL 9 heavy derived datasets (catTx filter/sort,
   176→  // dailyMap, chartData with 7-day moving avg, peakHour, timeOfDayMap,
   177→  // sourceList, dowMap, histogram, anomalies) are wrapped in a single
   178→  // useMemo so they only recompute when transactions, prevTransactions,
   179→  // cat, or selectedMonth changes. Previously every render iterated
   180→  // catTx 5+ times and constructed two `new Intl.DateTimeFormat(...)`
   181→  // instances inside per-tx loops. The formatter is now hoisted to
   182→  // module scope (JAKARTA_HOUR_FORMAT above) and reused.
   183→  const {
   184→    catTx,
   185→    chartData,
   186→    avgPerTx,
   187→    activeDays,
   188→    avgPerDay,
   189→    maxTx,
   190→    maxDay,
   191→    peakHour,
   192→    vsLastMonthPct,
   193→    vsLastMonthDir,
   194→    prevTotal,
   195→    timeOfDayMap,
   196→    maxTimeSlot,
   197→    topTimeSlot,
   198→    sourceList,
   199→    dailyAverage,
   200→    dowData,
   201→    dowTop,
   202→    histogram,
   203→    histMaxCount,
   204→    dominantBucket,
   205→    personalityTag,
   206→    anomalies,
   207→    primaryColor,
   208→  } = useMemo(() => {
   209→    // Filter transactions for this category
   210→    const catTx = transactions
   211→      .filter((t) => t.category === cat.name)
   212→      .sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime());
   213→
   214→    // Build daily breakdown for chart
   215→    const [yy, mm] = selectedMonth.split('-').map(Number);
   216→    const daysInMonth = new Date(yy, mm, 0).getDate();
   217→    const dailyMap = new Map<number, { total: number; count: number }>();
   218→    for (const tx of catTx) {
   219→      const day = parseInt(jakartaDateKey(new Date(tx.date)).slice(8, 10), 10);
   220→      const existing = dailyMap.get(day) ?? { total: 0, count: 0 };
   221→      existing.total += tx.amount || 0;
   222→      existing.count += 1;
   223→      dailyMap.set(day, existing);
   224→    }
   225→
   226→    let cumulative = 0;
   227→    const chartData: DailyData[] = [];
   228→    // Build raw daily totals first for moving average calculation
   229→    const rawTotals: number[] = [];
   230→    for (let d = 1; d <= daysInMonth; d++) {
   231→      const data = dailyMap.get(d);
   232→      rawTotals.push(data?.total ?? 0);
   233→    }
   234→    for (let d = 1; d <= daysInMonth; d++) {
   235→      const data = dailyMap.get(d);
   236→      cumulative += data?.total ?? 0;
   237→      const dateStr = `${selectedMonth}-${String(d).padStart(2, '0')}`;
   238→      const dateObj = new Date(Number(selectedMonth.split('-')[0]), Number(selectedMonth.split('-')[1]) - 1, d);
   239→      const dayLabel = dateObj.toLocaleDateString('en-US', { day: 'numeric', month: 'short' });
   240→      const axisLabel = d % 5 === 0 || d === 1 ? dayLabel : '';
   241→      // 7-day moving average centered on current day (3 days before + current + 3 after)
   242→      // Shows fluctuation trend instead of cumulative (which always goes up).
   243→      const windowStart = Math.max(0, d - 4); // 0-indexed, 3 days before
   244→      const windowEnd = Math.min(daysInMonth, d + 3); // 3 days after
   245→      const window = rawTotals.slice(windowStart, windowEnd);
   246→      const movingAvg = window.length > 0
   247→        ? Math.round(window.reduce((s, v) => s + v, 0) / window.length)
   248→        : 0;
   249→      chartData.push({
   250→        day: d,
   251→        date: dateStr,
   252→        label: axisLabel,
   253→        dateLabel: dayLabel,
   254→        total: data?.total ?? 0,
   255→        count: data?.count ?? 0,
   256→        cumulative,
   257→        movingAvg,
   258→      });
   259→    }
   260→
   261→    // Stats
   262→    const avgPerTx = catTx.length > 0 ? Math.round(cat.total / catTx.length) : 0;
   263→    const activeDays = dailyMap.size;
   264→    const avgPerDay = activeDays > 0 ? Math.round(cat.total / activeDays) : 0;
   265→    const maxTx = catTx.reduce((max, t) => (t.amount > max.amount ? t : max), catTx[0] ?? { amount: 0, description: '', date: '' });
   266→    const maxDay = Array.from(dailyMap.entries()).reduce(
   267→      (max, [day, v]) => (v.total > max.total ? { day, total: v.total } : max),
   268→      { day: 0, total: 0 }
   269→    );
   270→
   271→    // Peak hour pattern (uses hoisted JAKARTA_HOUR_FORMAT instead of
   272→    // creating a new Intl.DateTimeFormat per iteration)
   273→    const hourMap = new Map<number, number>();
   274→    for (const tx of catTx) {
   275→      const h = parseInt(JAKARTA_HOUR_FORMAT.format(new Date(tx.date)), 10) % 24;
   276→      hourMap.set(h, (hourMap.get(h) ?? 0) + 1);
   277→    }
   278→    const peakHour = Array.from(hourMap.entries()).reduce(
   279→      (max, [h, count]) => (count > max.count ? { hour: h, count } : max),
   280→      { hour: 0, count: 0 }
   281→    );
   282→
   283→    // ── B4: vs Last Month comparison ──────────────────────────────────
   284→    const prevCatTx = prevTransactions.filter((t) => t.category === cat.name);
   285→    const prevTotal = prevCatTx.reduce((s, t) => s + (t.amount || 0), 0);
   286→    const vsLastMonthPct = prevTotal > 0
   287→      ? Math.round(((cat.total - prevTotal) / prevTotal) * 100)
   288→      : null;
   289→    const vsLastMonthDir = vsLastMonthPct === null ? 'unknown'
   290→      : vsLastMonthPct > 0 ? 'up'
   291→      : vsLastMonthPct < 0 ? 'down'
   292→      : 'same';
   293→
   294→    // ── A2: Time-of-day distribution ──────────────────────────────────
   295→    const timeOfDayMap = {
   296→      pagi: { label: '🌅 Pagi (5-12)', count: 0, total: 0, hours: '05:00-11:59' },
   297→      siang: { label: '☀️ Siang (12-17)', count: 0, total: 0, hours: '12:00-16:59' },
   298→      sore: { label: '🌆 Sore (17-22)', count: 0, total: 0, hours: '17:00-21:59' },
   299→      malam: { label: '🌙 Malam (22-5)', count: 0, total: 0, hours: '22:00-04:59' },
   300→    };
   301→    for (const tx of catTx) {
   302→      const h = parseInt(JAKARTA_HOUR_FORMAT.format(new Date(tx.date)), 10) % 24;
   303→      if (h >= 5 && h < 12) { timeOfDayMap.pagi.count++; timeOfDayMap.pagi.total += tx.amount || 0; }
   304→      else if (h >= 12 && h < 17) { timeOfDayMap.siang.count++; timeOfDayMap.siang.total += tx.amount || 0; }
   305→      else if (h >= 17 && h < 22) { timeOfDayMap.sore.count++; timeOfDayMap.sore.total += tx.amount || 0; }
   306→      else { timeOfDayMap.malam.count++; timeOfDayMap.malam.total += tx.amount || 0; }
   307→    }
   308→    const maxTimeSlot = Math.max(
   309→      timeOfDayMap.pagi.count, timeOfDayMap.siang.count,
   310→      timeOfDayMap.sore.count, timeOfDayMap.malam.count, 1
   311→    );
   312→    const topTimeSlot = Object.entries(timeOfDayMap).reduce(
   313→      (max, [key, v]) => v.count > max.count ? { key, ...v } : max,
   314→      { key: '', label: '', count: 0, total: 0, hours: '' }
   315→    );
   316→
   317→    // ── C9: Source breakdown ──────────────────────────────────────────
   318→    const sourceMap = new Map<string, number>();
   319→    for (const tx of catTx) {
   320→      sourceMap.set(tx.source, (sourceMap.get(tx.source) ?? 0) + (tx.amount || 0));
   321→    }
   322→    const sourceList = Array.from(sourceMap.entries())
   323→      .map(([name, total]) => ({
   324→        name,
   325→        total,
   326→        percentage: cat.total > 0 ? Math.round((total / cat.total) * 100) : 0,
   327→      }))
   328→      .sort((a, b) => b.total - a.total);
   329→
   330→    // ── E15: Daily average for chart reference line ───────────────────
   331→    const dailyAverage = activeDays > 0 ? Math.round(cat.total / activeDays) : 0;
   332→
   333→    // ── A1: Day-of-week breakdown ─────────────────────────────────────
   334→    const dowMap = new Array(7).fill(0).map(() => ({ total: 0, count: 0 }));
   335→    for (const tx of catTx) {
   336→      const dateKey = jakartaDateKey(new Date(tx.date));
   337→      const d = new Date(dateKey + 'T00:00:00Z');
   338→      const dow = d.getUTCDay();
   339→      dowMap[dow].total += tx.amount || 0;
   340→      dowMap[dow].count += 1;
   341→    }
   342→    const dowTop = dowMap.reduce<{ idx: number; total: number; count: number }>(
   343→      (max, d, i) => (d.total > max.total ? { idx: i, total: d.total, count: d.count } : max),
   344→      { idx: 0, total: 0, count: 0 }
   345→    );
   346→    const dowData = dowMap.map((d, i) => ({
   347→      day: DOW_NAMES[i],
   348→      total: d.total,
   349→      count: d.count,
   350→      pct: cat.total > 0 ? Math.round((d.total / cat.total) * 100) : 0,
   351→    }));
   352→
   353→    // ── C8: Amount distribution histogram ─────────────────────────────
   354→    const histogram = (() => {
   355→      if (catTx.length === 0) return [];
   356→      // Determine bucket boundaries from max tx
   357→      const max = Math.max(...catTx.map((t) => t.amount || 0), 1);
   358→      // Create 5 buckets: 0-20%, 20-40%, 40-60%, 60-80%, 80-100% of max
   359→      const bucketSize = max / 5;
   360→      const buckets = new Array(5).fill(0).map((_, i) => ({
   361→        range: `${compactRupiahSafe(Math.round(i * bucketSize))}-${compactRupiahSafe(Math.round((i + 1) * bucketSize))}`,
   362→        count: 0,
   363→        total: 0,
   364→      }));
   365→      for (const tx of catTx) {
   366→        const amt = tx.amount || 0;
   367→        let idx = Math.floor(amt / bucketSize);
   368→        if (idx >= 5) idx = 4; // clamp max into last bucket
   369→        if (idx < 0) idx = 0;
   370→        buckets[idx].count++;
   371→        buckets[idx].total += amt;
   372→      }
   373→      return buckets;
   374→    })();
   375→    const histMaxCount = Math.max(...histogram.map((b) => b.count), 1);
   376→    const dominantBucket = histogram.reduce<{ idx: number; range: string; count: number; total: number }>(
   377→      (max, b, i) => (b.count > max.count ? { idx: i, range: b.range, count: b.count, total: b.total } : max),
   378→      { idx: 0, range: '', count: 0, total: 0 }
   379→    );
   380→
   381→    // ── D11: Personality tag ──────────────────────────────────────────
   382→    const personalityTag = (() => {
   383→      if (catTx.length === 0) return null;
   384→      // Check weekend dominance
   385→      const weekendCount = dowMap[0].count + dowMap[6].count; // Sun + Sat
   386→      const weekendPct = catTx.length > 0 ? weekendCount / catTx.length : 0;
   387→      // Check morning dominance
   388→      const morningPct = catTx.length > 0 ? timeOfDayMap.pagi.count / catTx.length : 0;
   389→      // Check frequency (transactions per active day)
   390→      const freq = activeDays > 0 ? catTx.length / activeDays : 0;
   391→      // Check if high spender (>30% of grand total)
   392→      const isHighSpender = cat.percentage >= 30;
   393→
   394→      if (isHighSpender) {
   395→        return { tag: 'Pengeluar Besar', emoji: '💸', desc: '>30% dari total pengeluaran bulan ini' };
   396→      }
   397→      if (freq >= 1.5) {
   398→        return { tag: 'Ritual Harian', emoji: '🔄', desc: 'Rata-rata lebih dari 1× per hari aktif' };
   399→      }
   400→      if (weekendPct >= 0.5) {
   401→        return { tag: 'Boros Akhir Pekan', emoji: '🎉', desc: `${Math.round(weekendPct * 100)}% transaksi di weekend` };
   402→      }
   403→      if (morningPct >= 0.7) {
   404→        return { tag: 'Ritual Pagi', emoji: '🌅', desc: `${Math.round(morningPct * 100)}% transaksi di pagi hari` };
   405→      }
   406→      if (catTx.length < 4) {
   407→        return { tag: 'Sekali-sekali', emoji: '🍃', desc: 'Kurang dari 4× per bulan' };
   408→      }
   409→      return { tag: 'Pengeluar Stabil', emoji: '⚖️', desc: 'Pola spending yang konsisten' };
   410→    })();
   411→
   412→    // ── D12: Anomaly detection ────────────────────────────────────────
   413→    const anomalies = (() => {
   414→      if (catTx.length < 3) return []; // need at least 3 for meaningful average
   415→      const amounts = catTx.map((t) => t.amount || 0);
   416→      const mean = amounts.reduce((a, b) => a + b, 0) / amounts.length;
   417→      const variance = amounts.reduce((s, v) => s + (v - mean) ** 2, 0) / amounts.length;
   418→      const sd = Math.sqrt(variance);
   419→      if (sd === 0) return [];
   420→      return catTx
   421→        .map((tx) => {
   422→          const z = (tx.amount - mean) / sd;
   423→          return { tx, zScore: Math.round(z * 100) / 100, mean: Math.round(mean) };
   424→        })
   425→        .filter((a) => a.zScore > 1.5) // 1.5σ above normal
   426→        .sort((a, b) => b.zScore - a.zScore);
   427→    })();
   428→
   429→    const primaryColor = cat.color || '#22c55e';
   430→
   431→    return {
   432→      catTx,
   433→      chartData,
   434→      avgPerTx,
   435→      activeDays,
   436→      avgPerDay,
   437→      maxTx,
   438→      maxDay,
   439→      peakHour,
   440→      vsLastMonthPct,
   441→      vsLastMonthDir,
   442→      prevTotal,
   443→      timeOfDayMap,
   444→      maxTimeSlot,
   445→      topTimeSlot,
   446→      sourceList,
   447→      dailyAverage,
   448→      dowData,
   449→      dowTop,
   450→      histogram,
   451→      histMaxCount,
   452→      dominantBucket,
   453→      personalityTag,
   454→      anomalies,
   455→      primaryColor,
   456→    };
   457→    // Deps note: `cat` is included (rather than just `cat.name`) so the
   458→    // memo recomputes whenever the parent's `categoryTotals` re-derives
   459→    // (e.g. if `getCategoryMeta` changes the emoji/color, or any cat.*
   460→    // field updates). `cat.name` alone would miss the getCategoryMeta
   461→    // edge case. Functionally equivalent to the audit's recommendation
   462→    // for the common case (transactions/selectedMonth change), stricter
   463→    // for the rare case.
   464→  }, [transactions, prevTransactions, cat, selectedMonth]);
   465→
   466→  return (
   467→    <div className="space-y-4 overflow-x-hidden">
   468→      {/* Breadcrumb + back + 1-click tx link */}
   469→      <div className="flex items-center gap-2">
   470→        <button
   471→          onClick={onBack}
   472→          className="flex items-center gap-1 text-xs font-medium text-muted-foreground hover:text-foreground transition-colors cursor-pointer"
   473→        >
   474→          <ChevronLeft className="h-4 w-4" />
   475→          Kategori
   476→        </button>
   477→        <ChevronLeft className="h-3 w-3 text-muted-foreground/50" />
   478→        <span className="text-xs font-medium text-foreground">{cat.emoji} {cat.name}</span>
   479→        {/* ONE-CLICK-8: ghost pill rata kanan — langsung ke transaksi
   480→            kategori ini (filter terpasang). */}
   481→        <button
   482→          type="button"
   483→          onClick={handleViewTransactions}
   484→          className="ml-auto inline-flex items-center gap-1 rounded-full px-2.5 py-1.5 min-h-8 text-xs font-semibold text-primary hover:bg-primary/10 active:scale-95 transition-all cursor-pointer focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-primary/40"
   485→          aria-label={`Lihat transaksi kategori ${cat.name}`}
   486→        >
   487→          Lihat transaksi
   488→          <ArrowUpRight className="h-3.5 w-3.5" aria-hidden="true" />
   489→        </button>
   490→      </div>
   491→
   492→      {/* Month picker */}
