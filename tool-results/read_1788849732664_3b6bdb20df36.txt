   221→  async function handleSubmit() {
   222→    if (!form.name.trim()) {
   223→      toast.error('Nama habit wajib diisi');
   224→      return;
   225→    }
   226→    setSubmitting(true);
   227→    try {
   228→      const payload = {
   229→        ...form,
   230→        reminder: form.reminder || null,
   231→        endDate: form.endDate || null,
   232→        notes: form.notes || null,
   233→        // BUG-3 fix: clamp target to 1 — UI only supports binary completion.
   234→        // (Also serves as a safety net for legacy habits edited with target>1.)
   235→        // PHASE3-HABIT: for "amount" habits (daily goal with numeric target),
   236→        // allow target > 1 (up to 1000 — see the form's max attribute).
   237→        target: form.habitType === 'amount' ? Math.min(1000, Math.max(1, form.target || 1)) : 1,
   238→        // targetType is preserved from the form (default 'daily' for new
   239→        // habits; existing habits keep their value). Non-daily options are
   240→        // disabled in the dropdown so users can't pick an unsupported mode,
   241→        // but we don't overwrite legacy values on save (BUG-14 minimal fix).
   242→        targetTime: form.targetTime || null,
   243→        trackLastDone: form.trackLastDone,
   244→        lastDoneInterval: form.lastDoneInterval || null,
   245→        groupId: form.groupId || null,
   246→        // PHASE1-HABIT: vacation mode + end date. Empty string → null so the
   247→        // API stores null (indefinite vacation) rather than an empty date.
   248→        vacationMode: form.vacationMode,
   249→        vacationEnd: form.vacationEnd || null,
   250→        // PHASE3-HABIT: habit type ("normal" | "avoid" | "amount"). Controls
   251→        // how the daily-tracker interprets the checkbox and how the card is
   252→        // displayed (avoid → red relapse state, amount → progress bar).
   253→        habitType: form.habitType,
   254→      };
   255→
   256→      if (editingId) {
   257→        // Update
   258→        const res = await fetch(`/api/habits/${editingId}`, {
   259→          method: 'PUT',
   260→          headers: { 'Content-Type': 'application/json' },
   261→          body: JSON.stringify(payload),
   262→        });
   263→        if (!res.ok) throw new Error('Failed to update habit');
   264→        // Optimistic update
   265→        queryClient.setQueryData<Habit[]>(['habits'], (prev = []) =>
   266→          prev.map((h) =>
   267→            h.id === editingId ? { ...h, ...payload, updatedAt: new Date().toISOString() } : h
   268→          )
   269→        );
   270→        toast.success('Habit berhasil diperbarui');
   271→      } else {
   272→        // Create
   273→        const res = await fetch('/api/habits', {
   274→          method: 'POST',
   275→          headers: { 'Content-Type': 'application/json' },
   276→          body: JSON.stringify(payload),
   277→        });
   278→        if (!res.ok) throw new Error('Failed to create habit');
   279→        const newHabit = await res.json();
   280→        // Optimistic update
   281→        queryClient.setQueryData<Habit[]>(['habits'], (prev = []) => [...prev, newHabit]);
   282→        toast.success('Habit berhasil dibuat');
   283→      }
   284→
   285→      setDialogOpen(false);
   286→      triggerRefresh();
   287→    } catch {
   288→      toast.error(editingId ? 'Gagal memperbarui habit' : 'Gagal membuat habit');
   289→      // Re-fetch on failure
   290→      invalidateHabits();
   291→    } finally {
   292→      setSubmitting(false);
   293→    }
   294→  }

   315→  async function handleToggleStatus(h: Habit) {
   316→    const newStatus = h.status === 'active' ? 'paused' : 'active';
   317→    const statusLabel = newStatus === 'paused' ? 'dijeda' : 'dilanjutkan';
   318→    // Optimistic
   319→    queryClient.setQueryData<Habit[]>(['habits'], (prev = []) =>
   320→      prev.map((x) => (x.id === h.id ? { ...x, status: newStatus } : x))
   321→    );
   322→    try {
   323→      const res = await fetch(`/api/habits/${h.id}`, {
   324→        method: 'PUT',
   325→        headers: { 'Content-Type': 'application/json' },
   326→        body: JSON.stringify({ status: newStatus }),
   327→      });
   328→      if (!res.ok) throw new Error();
   329→      toast.success(`Habit ${statusLabel}`);
   330→      triggerRefresh();
   331→    } catch {
   332→      toast.error(`Gagal ${statusLabel} habit`);
   333→      invalidateHabits();
   334→    }
   335→  }
