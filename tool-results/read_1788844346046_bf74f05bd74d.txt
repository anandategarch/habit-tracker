    56→    56→// PERF-FIX (FIX-TIER3 / Fix 14): Format time from transaction date.
    57→    57→// Hoisted to module scope — this function is stateless and was previously
    58→    58→// re-created on every render. MUST use timeZone: 'Asia/Jakarta' explicitly
    59→    59→// (without it, toLocaleTimeString uses the runtime's default TZ — UTC on
    60→    60→// Vercel serverless, or the user's browser TZ locally). This caused the
    61→    61→// time shown here to disagree with the time shown in the Daily Recap's
    62→    62→// hourly heatmap + transactions list (which both use timeZone:
    63→    63→// 'Asia/Jakarta'). Same code path as daily-recap.tsx's formatTxTime() —
    64→    64→// keep them in sync.
    65→    65→function formatTime(dateStr: string) {
    66→    66→  try {
    67→    67→    const d = new Date(dateStr);
    68→    68→    return d.toLocaleTimeString('id-ID', {
    69→    69→      timeZone: 'Asia/Jakarta',
    70→    70→      hour: '2-digit',
    71→    71→      minute: '2-digit',
    72→    72→    });
    73→    73→  } catch {
    74→    74→    return '';
    75→    75→  }
    76→    76→}
    77→    77→
    78→    78→// PERF-FIX (FIX-TIER3 / Fix 14): Flat row representation for the virtualized
    79→    79→// list. The original component rendered a nested structure (groups → txs),
    80→    80→// which forced the browser to allocate a DOM node for every transaction
    81→    81→// even when off-screen. For months with 100+ transactions this meant
    82→    82→// hundreds of nodes living in the layout. Flattening groups into a single
    83→    83→// row list lets @tanstack/react-virtual only render the visible window.
    84→    84→type FlatRow =
    85→    85→  | { kind: 'header'; group: GroupedTransaction }
    86→    86→  | { kind: 'tx'; tx: Transaction; txIdx: number };
    87→    87→
    88→    88→// PERF-FIX: estimateSize callbacks must be stable (not re-created each
    89→    89→// render) — the virtualizer uses referential equality to decide whether
    90→    90→// to re-measure. Hoisting to module scope also avoids a fresh closure
    91→    91→// per render, which would otherwise cause the virtualizer to reset its
    92→    92→// measurements on every render of this component.
    93→    93→function estimateRowSize(row: FlatRow | undefined): number {
    94→    94→  if (!row) return 80;
    95→    95→  // Header rows are short (single-line pill). Transaction rows are taller
    96→    96→  // (icon + 2-line content + amount + action buttons).
    97→    97→  // FIX: updated from 92 → 122 to match actual card height after spacing fix
    98→    98→  // (16px padding × 2 + 78px content + 12px margin-bottom = 122px).
    99→    99→  // Also enabled measureElement below for self-healing dynamic measurement.
   100→   100→  return row.kind === 'header' ? 36 : 122;
   101→   101→}
