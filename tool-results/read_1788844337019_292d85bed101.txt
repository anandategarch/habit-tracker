   247→  const handleDragEnd = useCallback(
   248→    async (event: DragEndEvent) => {

   252→      const newIndex = activeHabits.findIndex((h) => h.id === over.id);

   259→      reordered.forEach((h, idx) => {

   261→      });

   265→      const activeIds = new Set(reordered.map((h) => h.id));

   268→        ...reordered.map((h, idx) => ({ ...h, order: idx })),

   280→              method: 'PUT',
   281→              headers: { 'Content-Type': 'application/json' },

   284→          ),

   292→        toast.error('Gagal menyimpan urutan');
   293→        // Revert to server truth.
   294→        await queryClient.invalidateQueries({ queryKey: ['habits'] });
   295→        setLocalHabitsOverride(null);
   296→      }
   297→    },
   298→    [activeHabits, habits, queryClient],
   299→  );

   397→      setCompletedAtMap(atMap);

   414→    }

   417→    // Bailing here avoids: (a) painting stale completion maps over the
   418→    // current date's state, (b) flipping cachedMonthRef back to the stale
   419→    // month (which would only cost a redundant re-fetch later, but still).
   420→    if (isCancelled?.()) return;
   421→
   422→    const monthCache: Record<string, HabitLog[]> = {};
   423→    const map: Record<string, boolean> = {};
   424→    const atMap: Record<string, string> = {};
   425→
   426→    active.forEach((habit) => {

   428→      monthCache[habit.id] = logs;
   429→      const dayLog = logs.find((l) => toDateString(l.date) === date);
   430→      map[habit.id] = dayLog?.completed ?? false;
   431→      if (dayLog?.completedAt) {
   432→        atMap[habit.id] = formatJakartaTime(dayLog.completedAt);
   433→      }
   434→    });
   435→
   436→    monthLogsCacheRef.current[month] = monthCache;
   437→    cachedMonthRef.current = month;
   438→    // BUG-16 fix: record the refreshKey that populated this cache so a future
   439→    // refreshKey change invalidates the cache.
   440→    cachedRefreshKeyRef.current = refreshKey;
   441→    setCompletionMap(map);
   442→    setCompletedAtMap(atMap);
   443→  };

   445→  // ---- debounced save (notes only) ----

   455→        pendingSaveRef.current = null;
   456→        if (!pending) return;
   457→        try {
   458→          await fetch('/api/daily-logs', {
   459→            method: 'POST',
   460→            headers: { 'Content-Type': 'application/json' },
   461→            body: JSON.stringify(pending),
   462→          });
   463→          queryClient.invalidateQueries({ queryKey: ['daily-logs', pending.date] });
   464→        } catch {
   465→          toast.error('Gagal menyimpan catatan');
   466→        }
   467→      }, 600);
   468→    },
   469→    [selectedDate, queryClient],
   470→  );

   474→      setNotes(html);
   475→      debouncedSave({ notes: html });
   476→    },
   477→    [debouncedSave],

   482→  // tags etc. — which would be misleading. Strip tags to get the user-visible
   483→  // length. Returns 0 for empty/whitespace-only content.
   484→  const notesCharCount = useMemo(() => {
   485→    if (!notes) return 0;
   486→    // Quick + dirty tag stripper — sufficient for the count display only.
   487→    // (For rendering, the TipTap editor handles its own HTML safely.)
   488→    return notes.replace(/<[^>]+>/g, '').replace(/&nbsp;/g, ' ').trim().length;
   489→  }, [notes]);

   491→  // ---- handlers ----
   492→  // Ref to track the element that triggered a habit completion (for confetti position).
   493→  // Set in handleHabitCheck, read in toggleHabit after successful API response.
   494→  const confettiElRef = useRef<HTMLElement | null>(null);
   495→
   496→  // PERF-REACT-1 fix: toggleHabit is declared BEFORE handleHabitCheck and
   497→  // handleTimeDialogSubmit (which call it) so the useCallback deps arrays
   498→  // can reference it without temporal-dead-zone errors. Reads
   499→  // `completionMap` via `completionMapRef.current` (not directly) so the
   500→  // callback identity stays stable across toggles — this is what lets
