    66→    66→/** Live preview theme colors without saving to DB */
    67→    67→function previewTheme(primary: string, secondary: string, theme: string) {
    68→    68→ const isDark = theme === 'dark';
    69→    69→ applyThemeMode(isDark);
    70→    70→
    71→    71→ if (
    72→    72→   primary.toLowerCase() === '#22c55e' &&
    73→    73→   secondary.toLowerCase() === '#10b981'
    74→    74→ ) {
    75→    75→   resetThemeColors();
    76→    76→ } else {
    77→    77→   applyThemeColors(primary, secondary, isDark);
    78→    78→ }
    79→    79→}
