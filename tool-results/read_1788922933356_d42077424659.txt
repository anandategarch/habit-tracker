   318→ // ── PHASE4-POLISH: drag-to-reorder (@dnd-kit) ──────────────────────────
   319→ // Sensors + handlers are declared HERE (after `activeHabits` is defined)

   322→ // Sensors: PointerSensor (mouse), TouchSensor (mobile drag — required for

   334→
   335→ const handleDragEnd = useCallback(
   336→   async (event: DragEndEvent) => {
   337→     const { active, over } = event;
   338→     if (!over || active.id === over.id) return;
   339→     const oldIndex = activeHabits.findIndex((h) => h.id === active.id);
   340→     const newIndex = activeHabits.findIndex((h) => h.id === over.id);
   341→     if (oldIndex < 0 || newIndex < 0) return;
   342→
   343→     const reordered = arrayMove(activeHabits, oldIndex, newIndex);
   344→     // Re-assign `order` so the new array position matches the DB order
   345→     // (0..N-1 across active habits). Collect only the diffs to PUT.
   346→     const updates: { id: string; sortOrder: number }[] = [];
   347→     reordered.forEach((h, idx) => {
   348→       if (h.sortOrder !== idx) updates.push({ id: h.id, sortOrder: idx });
   349→     });
   350→
   351→     // Optimistic local override: reordered active habits (with new order
   352→     // field) followed by the unchanged paused/archived habits.
   353→     const activeIds = new Set(reordered.map((h) => h.id));
   354→     const nonActive = habits.filter((h) => !activeIds.has(h.id));
   355→     const reorderedAll: Habit[] = [
   356→       ...reordered.map((h, idx) => ({ ...h, sortOrder: idx })),
   357→       ...nonActive,
   358→     ];
   359→     setLocalHabitsOverride(reorderedAll);
   360→
   361→     // Persist each changed habit's order via PUT /api/habits/[id].
   362→     // Fire-and-forget in parallel; invalidate the query on settle so the
   363→     // server-side truth is re-fetched (and the local override cleared).
   364→     try {
   365→       await Promise.all(
   366→         updates.map((u) =>
   367→           fetch(`/api/habits/${u.id}`, {
   368→             method: 'PUT',
   369→             headers: { 'Content-Type': 'application/json' },
   370→             body: JSON.stringify({ sortOrder: u.sortOrder }),
   371→           }),
   372→         ),
   373→       );
   374→       await queryClient.invalidateQueries({ queryKey: ['habits'] });
   375→       // Brief delay so the refetch lands before we drop the override —
   376→       // otherwise a re-render with the stale query cache could flicker
   377→       // back to the old order for one frame.
   378→       setTimeout(() => setLocalHabitsOverride(null), 200);
   379→     } catch {
   380→       toast.error('Gagal menyimpan urutan');
   381→       // Revert to server truth.
   382→       await queryClient.invalidateQueries({ queryKey: ['habits'] });
   383→       setLocalHabitsOverride(null);
   384→     }
   385→   },
   386→   [activeHabits, habits, queryClient],
   387→ );
   388→

   443→     // PHASE1-HABIT: pass vacationMode so vacationing habits' streaks don't

   457→ // ---- fetch completions (month-cached) ----
   458→ // BUGHUNT-ROUND2 RACE-1: `isCancelled` guard (passed by the loading
   459→ // effect) prevents a SLOWER stale fetch (e.g. for the previous date/
   460→ // month after rapid navigation) from overwriting the state of the
   461→ // NEWER fetch that already resolved. Previously only `setLoading` was

   464→ const fetchCompletions = async (
   465→   habitList: Habit[],
   466→   date: string,
   467→   isCancelled?: () => boolean,
   468→ ) => {
   469→   const month = date.slice(0, 7);
   470→
   471→   // BUG-16 fix: include refreshKey in the cache hit check. Without this,
   472→   // changing refreshKey (e.g. after creating a habit) would still hit the
   473→   // stale cache and never re-fetch.
   474→   if (
   475→     cachedMonthRef.current === month &&
   476→     monthLogsCacheRef.current[month] &&
   477→     cachedRefreshKeyRef.current === refreshKey
   478→   ) {
   479→     const cache = monthLogsCacheRef.current[month];
   480→     const map: Record<string, boolean> = {};
   481→     const atMap: Record<string, string> = {};
   482→     const valueMap: Record<string, number> = {};
   483→     habitList
