   219→  async function handleSubmit() {
   220→    if (!form.name.trim()) {
   221→      toast.error('Nama habit wajib diisi');
   222→      return;
   223→    }
   224→    setSubmitting(true);
   225→    try {
   226→      const payload = {
   227→        ...form,
   228→        reminder: form.reminder || null,
   229→        endDate: form.endDate || null,
   230→        notes: form.notes || null,
   231→        // BUG-3 fix: clamp target to 1 — UI only supports binary completion.
   232→        // (Also serves as a safety net for legacy habits edited with target>1.)
   233→        // PHASE3-HABIT: for "amount" habits (daily goal with numeric target),
   234→        // allow target > 1 (up to 1000 — see the form's max attribute).
   235→        target: form.habitType === 'amount' ? Math.min(1000, Math.max(1, form.target || 1)) : 1,
   236→        // targetType is preserved from the form (default 'daily' for new
   237→        // habits; existing habits keep their value). Non-daily options are
   238→        // disabled in the dropdown so users can't pick an unsupported mode,
   239→        // but we don't overwrite legacy values on save (BUG-14 minimal fix).
   240→        targetTime: form.targetTime || null,
   241→        trackLastDone: form.trackLastDone,
   242→        lastDoneInterval: form.lastDoneInterval || null,
   243→        groupId: form.groupId || null,
   244→        // PHASE1-HABIT: vacation mode + end date. Empty string → null so the
   245→        // API stores null (indefinite vacation) rather than an empty date.
   246→        vacationMode: form.vacationMode,
   247→        vacationEnd: form.vacationEnd || null,
   248→        // PHASE3-HABIT: habit type ("normal" | "avoid" | "amount"). Controls
   249→        // how the daily-tracker interprets the checkbox and how the card is
   250→        // displayed (avoid → red relapse state, amount → progress bar).
   251→        habitType: form.habitType,
   252→      };
   253→
   254→      if (editingId) {
   255→        // Update
   256→        const res = await fetch(`/api/habits/${editingId}`, {
   257→          method: 'PUT',
   258→          headers: { 'Content-Type': 'application/json' },
   259→          body: JSON.stringify(payload),
   260→        });
   261→        if (!res.ok) throw new Error('Failed to update habit');
   262→        // Optimistic update
   263→        queryClient.setQueryData<Habit[]>(['habits'], (prev = []) =>
   264→          prev.map((h) =>
   265→            h.id === editingId ? { ...h, ...payload, updatedAt: new Date().toISOString() } : h
   266→          )
   267→        );
   268→        toast.success('Habit berhasil diperbarui');
   269→      } else {
   270→        // Create
   271→        const res = await fetch('/api/habits', {
   272→          method: 'POST',
   273→          headers: { 'Content-Type': 'application/json' },
   274→          body: JSON.stringify(payload),
   275→        });
   276→        if (!res.ok) throw new Error('Failed to create habit');
   277→        const newHabit = await res.json();
   278→        // Optimistic update
   279→        queryClient.setQueryData<Habit[]>(['habits'], (prev = []) => [...prev, newHabit]);
   280→        toast.success('Habit berhasil dibuat');
   281→      }
   282→
   283→      setDialogOpen(false);
   284→      triggerRefresh();
   285→    } catch {
   286→      toast.error(editingId ? 'Gagal memperbarui habit' : 'Gagal membuat habit');
   287→      // Re-fetch on failure
   288→      invalidateHabits();
   289→    } finally {
   290→      setSubmitting(false);
   291→    }
   292→  }

   313→  async function handleToggleStatus(h: Habit) {
   314→    const newStatus = h.status === 'active' ? 'paused' : 'active';
   315→    const statusLabel = newStatus === 'paused' ? 'dijeda' : 'dilanjutkan';
   316→    // Optimistic

   318→      prev.map((x) => (x.id === h.id ? { ...x, status: newStatus } : x))
   319→    );
   320→    try {
   321→      const res = await fetch(`/api/habits/${h.id}`, {
   322→        method: 'PUT',
   323→        headers: { 'Content-Type': 'application/json' },
   324→        body: JSON.stringify({ status: newStatus }),
   325→      });
   326→      if (!res.ok) throw new Error();
   327→      toast.success(`Habit ${statusLabel}`);
   328→      triggerRefresh();
   329→    } catch {
   330→      toast.error(`Gagal ${statusLabel} habit`);

   332→    }
