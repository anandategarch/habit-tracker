   271→  const handleDragEnd = useCallback(
   272→    async (event: DragEndEvent) => {
   273→      const { active, over } = event;
   274→      if (!over || active.id === over.id) return;
   275→      const oldIndex = activeHabits.findIndex((h) => h.id === active.id);
   276→      const newIndex = activeHabits.findIndex((h) => h.id === over.id);
   277→      if (oldIndex < 0 || newIndex < 0) return;
   278→
   279→      const reordered = arrayMove(activeHabits, oldIndex, newIndex);
   280→      // Re-assign `order` so the new array position matches the DB order
   281→      // (0..N-1 across active habits). Collect only the diffs to PUT.
   282→      const updates: { id: string; order: number }[] = [];
   283→      reordered.forEach((h, idx) => {
   284→        if (h.order !== idx) updates.push({ id: h.id, order: idx });
   285→      });
   286→
   287→      // Optimistic local override: reordered active habits (with new order
   288→      // field) followed by the unchanged paused/archived habits.
   289→      const activeIds = new Set(reordered.map((h) => h.id));
   290→      const nonActive = habits.filter((h) => !activeIds.has(h.id));
   291→      const reorderedAll: Habit[] = [
   292→        ...reordered.map((h, idx) => ({ ...h, order: idx })),
   293→        ...nonActive,
   294→      ];
   295→      setLocalHabitsOverride(reorderedAll);
   296→
   297→      // Persist each changed habit's order via PUT /api/habits/[id].

   299→      // server-side truth is re-fetched (and the local override cleared).

   301→        await Promise.all(
   302→          updates.map((u) =>
   303→            fetch(`/api/habits/${u.id}`, {
   304→              method: 'PUT',
   305→              headers: { 'Content-Type': 'application/json' },
   306→              body: JSON.stringify({ order: u.order }),
   307→            }),
   308→          ),
   309→        );
   310→        await queryClient.invalidateQueries({ queryKey: ['habits'] });
   311→        // Brief delay so the refetch lands before we drop the override —
   312→        // otherwise a re-render with the stale query cache could flicker
   313→        // back to the old order for one frame.
   314→        setTimeout(() => setLocalHabitsOverride(null), 200);
   315→      } catch {
   316→        toast.error('Gagal menyimpan urutan');
   317→        // Revert to server truth.
   318→        await queryClient.invalidateQueries({ queryKey: ['habits'] });
   319→        setLocalHabitsOverride(null);
   320→      }
   321→    },
   322→    [activeHabits, habits, queryClient],
   323→  );
   324→
   325→  /** Toggle drag mode. When enabling, force viewFilter to "all" so every

   385→  // ---- fetch completions (month-cached) ----
   386→  // BUGHUNT-ROUND2 RACE-1: `isCancelled` guard (passed by the loading
   387→  // effect) prevents a SLOWER stale fetch (e.g. for the previous date/

   389→  // NEWER fetch that already resolved. Previously only `setLoading` was

   392→  const fetchCompletions = async (
   393→    habitList: Habit[],
   394→    date: string,
   395→    isCancelled?: () => boolean,
   396→  ) => {
   397→    const month = date.slice(0, 7);
   398→

   403→      cachedMonthRef.current === month &&
   404→      monthLogsCacheRef.current[month] &&
   405→      cachedRefreshKeyRef.current === refreshKey
   406→    ) {
   407→      const cache = monthLogsCacheRef.current[month];
   408→      const map: Record<string, boolean> = {};
   409→      const atMap: Record<string, string> = {};
   410→      habitList
   411→        .filter((h) => h.status === 'active')
   412→        .forEach((h) => {
   413→          const logs = cache[h.id] || [];
   414→          const dayLog = logs.find((l) => toDateString(l.date) === date);
   415→          map[h.id] = dayLog?.completed ?? false;
   416→          if (dayLog?.completedAt) {
   417→            atMap[h.id] = formatJakartaTime(dayLog.completedAt);
   418→          }
   419→        });
   420→      setCompletionMap(map);
   421→      setCompletedAtMap(atMap);
   422→      return;
   423→    }
   424→
   425→    const active = habitList.filter((h) => h.status === 'active');
   426→    const ids = active.map((h) => h.id);
   427→
   428→    let groupedLogs: Record<string, HabitLog[]> = {};
   429→    try {
   430→      const res = await fetch(
   431→        `/api/habits/batch-logs?month=${month}&habitIds=${ids.join(',')}`,
   432→      );
   433→      if (res.ok) {
   434→        groupedLogs = await res.json();
   435→      }
   436→    } catch {
   437→      // fall through to empty defaults
   438→    }
   439→
   440→    // RACE-1: a newer fetch may have resolved while this one was in flight.
   441→    // Bailing here avoids: (a) painting stale completion maps over the
   442→    // current date's state, (b) flipping cachedMonthRef back to the stale
   443→    // month (which would only cost a redundant re-fetch later, but still).
   444→    if (isCancelled?.()) return;
   445→
   446→    const monthCache: Record<string, HabitLog[]> = {};
   447→    const map: Record<string, boolean> = {};
   448→    const atMap: Record<string, string> = {};
   449→
   450→    active.forEach((habit) => {
   451→      const logs = groupedLogs[habit.id] || [];
   452→      monthCache[habit.id] = logs;
   453→      const dayLog = logs.find((l) => toDateString(l.date) === date);
   454→      map[habit.id] = dayLog?.completed ?? false;
   455→      if (dayLog?.completedAt) {
   456→        atMap[habit.id] = formatJakartaTime(dayLog.completedAt);
   457→      }
   458→    });
   459→
   460→    monthLogsCacheRef.current[month] = monthCache;
   461→    cachedMonthRef.current = month;
   462→    // BUG-16 fix: record the refreshKey that populated this cache so a future
   463→    // refreshKey change invalidates the cache.
   464→    cachedRefreshKeyRef.current = refreshKey;
   465→    setCompletionMap(map);
   466→    setCompletedAtMap(atMap);
   467→  };
   468→
   469→  // ---- debounced save (notes only) ----
   470→  const debouncedSave = useCallback(
   471→    (patch: { notes?: string }) => {
   472→      if (saveTimerRef.current) clearTimeout(saveTimerRef.current);
   473→      // BUG-19 fix: stash the pending patch (with the current selectedDate)
   474→      // so the unmount handler can fire-and-forget it. Previously the timer
   475→      // was just cancelled on unmount, losing the last <600ms of typing.
   476→      pendingSaveRef.current = { date: selectedDate, ...patch };
   477→      saveTimerRef.current = setTimeout(async () => {
   478→        const pending = pendingSaveRef.current;
   479→        pendingSaveRef.current = null;
   480→        if (!pending) return;
   481→        try {
   482→          await fetch('/api/daily-logs', {
   483→            method: 'POST',
   484→            headers: { 'Content-Type': 'application/json' },
   485→            body: JSON.stringify(pending),
   486→          });
   487→          queryClient.invalidateQueries({ queryKey: ['daily-logs', pending.date] });
   488→        } catch {
   489→          toast.error('Gagal menyimpan catatan');
   490→        }
   491→      }, 600);
   492→    },
   493→    [selectedDate, queryClient],
   494→  );
   495→
   496→  const handleNotesChange = useCallback(
