   487→        tagsArr.some((t) => t.toLowerCase().includes(term));
   488→      if (!matches) return false;
   489→    }
   490→    return true;
   491→  }), [transactions, txFilter]);

   526→      else { currentGroup.totalExpense += tx.amount; currentGroup.net -= tx.amount; }
   527→    }
