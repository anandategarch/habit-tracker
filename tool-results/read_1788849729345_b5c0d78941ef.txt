    42→  const [whatIfReduction, setWhatIfReduction] = useState(0);
    43→  // Insight per kategori period tab: 'month' (current month) or 'alltime'.

    50→    queryKey: ['finance', 'daily-recap'],
    51→    queryFn: async () => {
    52→      const res = await fetch('/api/finance/daily-recap');

    68→  // BUG-2 fix: capture isError + refetch so the UI can show an error state

    70→  // the POST fails (network error, 500, etc.).
    71→  const allTimeCategories = recap?.today.categoryStats.map((c) => c.name) ?? [];
    72→  const {
    73→    data: allTimeStats,
    74→    isError: allTimeIsError,
    75→    refetch: refetchAllTime,
    76→  } = useQuery<Record<string, { maxTransaction: number; avgTransaction: number; maxDaily: number; avgDaily: number }>>({

    78→    queryFn: async () => {

    80→        method: 'POST',

    85→      return res.json();
    86→    },

    89→    retry: 1,

    91→
    92→  // Mutation: save daily budget target via PUT /api/settings
    93→  const saveBudgetMutation = useMutation({
    94→    mutationFn: async (target: number) => {
    95→      const res = await fetch('/api/settings', {
    96→        method: 'PUT',
    97→        headers: { 'Content-Type': 'application/json' },
    98→        body: JSON.stringify({ dailyBudgetTarget: target }),
    99→      });
   100→      if (!res.ok) throw new Error('Failed to save budget');
   101→      return res.json();
   102→    },
   103→    onSuccess: () => {
   104→      // Invalidate both daily-recap (re-fetch with new budget) and settings
   105→      queryClient.invalidateQueries({ queryKey: ['finance', 'daily-recap'] });
   106→      queryClient.invalidateQueries({ queryKey: ['settings'] });
   107→      toast.success('Budget harian disimpan');
   108→      setBudgetDialogOpen(false);
   109→    },
   110→    onError: () => {
   111→      toast.error('Gagal menyimpan budget');
   112→    },
   113→  });
   114→
   115→  // Mutation: save projection category selection via PUT /api/settings.
   116→  // Sends the full array of selected category names (empty = all categories).
   117→  // The API dedupes/sorts before storing, so we just send the raw selection.
   118→  //
   119→  // OPTIMISTIC UPDATE: the daily-recap query is heavy (30-day transaction
   120→  // fetch + 15 metric computations). Without optimistic update, each dropdown
   121→  // change caused a visible lag — the dropdown value wouldn't update until
   122→  // the PUT + refetch round-trip completed (300-800ms). Now we immediately
   123→  // patch the cached daily-recap's `projectionCategoryNames` so the dropdown
   124→  // reflects the new selection instantly; the actual projection NUMBER
   125→  // updates when the refetch completes (still server-side, still accurate).
   126→  // On error, we roll back to the previous value.
   127→  const saveProjectionCategoriesMutation = useMutation({
   128→    mutationFn: async (categoryNames: string[]) => {
   129→      const res = await fetch('/api/settings', {
   130→        method: 'PUT',
   131→        headers: { 'Content-Type': 'application/json' },
   132→        body: JSON.stringify({ projectionCategoryIds: categoryNames }),
   133→      });
   134→      if (!res.ok) throw new Error('Failed to save projection categories');
   135→      return res.json();
   136→    },
   137→    onMutate: async (categoryNames: string[]) => {
   138→      // Cancel any outgoing refetches so they don't overwrite our optimistic
   139→      // update.
   140→      await queryClient.cancelQueries({ queryKey: ['finance', 'daily-recap'] });
   141→      // Snapshot the previous value for rollback.
   142→      const previousRecap = queryClient.getQueryData<DailyRecapData>(['finance', 'daily-recap']);
   143→      // Optimistically patch the cached recap's projectionCategoryNames +
   144→      // projectionIsFiltered so the dropdown updates instantly.
   145→      //
   146→      // BUG-3 fix: Also reset whatIfBase to 0. The what-if slider computes
   147→      // its result from `whatIfBase` (month-to-date spend of the CURRENT
   148→      // basis). Without resetting it, the slider would use the STALE
   149→      // whatIfBase (from the old filter) for 300-800ms until the refetch
   150→      // completes — producing wrong numbers. Setting whatIfBase to 0
   151→      // hides the slider via its `whatIfBase > 0` guard, so the user sees
   152→      // a clean "loading" state instead of wrong numbers. The slider
   153→      // reappears with the correct whatIfBase once the refetch completes.
   154→      if (previousRecap) {
   155→        queryClient.setQueryData<DailyRecapData>(['finance', 'daily-recap'], {
   156→          ...previousRecap,
   157→          predictions: {
   158→            ...previousRecap.predictions,
   159→            projectionCategoryNames: categoryNames,
   160→            projectionIsFiltered: categoryNames.length > 0,
   161→            whatIfBase: 0,
   162→          },
   163→        });
   164→      }
   165→      return { previousRecap };
   166→    },
   167→    onSuccess: () => {
   168→      queryClient.invalidateQueries({ queryKey: ['finance', 'daily-recap'] });
   169→      queryClient.invalidateQueries({ queryKey: ['settings'] });
   170→    },
   171→    onError: (_err, _vars, context) => {
   172→      // Roll back to the previous cached value on error.
   173→      if (context?.previousRecap) {
   174→        queryClient.setQueryData(['finance', 'daily-recap'], context.previousRecap);
   175→      }
   176→      toast.error('Gagal menyimpan pilihan kategori');
   177→    },
   178→  });
   179→
   180→  /**
   181→   * Set the projection basis to a SINGLE category (single-select dropdown).
   182→   * - Empty string = use all categories (default, "Semua" option).
   183→   * - Non-empty = project based on that one category only.
   184→   *
   185→   * Single-select per the user's request — multi-select chips caused lag
   186→   * (each toggle = PUT + heavy refetch) and were overkill for the use case
   187→   * ("what's my month-end projection if I only consider Food spending?").
   188→   *
   189→   * BUG-3 fix: Reset whatIfReduction to 0 so the slider starts fresh on
   190→   * the new basis. Without this, the slider would show a stale reduction
   191→   * percentage (e.g., 20%) applied to the new basis — confusing.
   192→   */
   193→  function setProjectionCategory(categoryName: string) {
   194→    if (saveProjectionCategoriesMutation.isPending) return;
   195→    const next = categoryName ? [categoryName] : [];
   196→    setWhatIfReduction(0);
   197→    saveProjectionCategoriesMutation.mutate(next);
   198→  }
   199→
   200→  /** Open the budget dialog, pre-filling the input with the current target. */
   201→  function openBudgetDialog(currentTarget: number | null) {
   202→    // Format with thousand separators so the prefill matches what the
   203→    // onChange handler would produce (was showing "100000" instead of "100.000").
   204→    const formatted = currentTarget
   205→      ? String(currentTarget).replace(/\B(?=(\d{3})+(?!\d))/g, '.')
   206→      : '';
   207→    setBudgetInput(formatted);
   208→    setBudgetDialogOpen(true);
   209→  }
   210→
   211→  /** Parse the formatted input ("100.000" → 100000) and save. */
   212→  function handleSaveBudget() {

   214→    // could fire multiple concurrent mutations (race condition in PUT /api/settings).
   215→    if (saveBudgetMutation.isPending) return;
   216→    // Strip non-digits (handles "100.000", "100,000", " 100000 ")
   217→    const digits = budgetInput.replace(/[^\d]/g, '');
   218→    const target = digits ? parseInt(digits, 10) : 0;
   219→    if (target > 100_000_000) {
   220→      toast.error('Maksimal Rp 100.000.000');
   221→      return;
   222→    }
   223→    saveBudgetMutation.mutate(target);
   224→  }
   225→
   226→  /** Remove the daily budget (set to 0 → ring hidden). */
   227→  function handleRemoveBudget() {

   236→        <div className="p-4 flex items-center gap-3">
   237→          <div className="flex items-center justify-center w-8 h-8 rounded-lg bg-destructive/10 text-destructive dark:bg-destructive/15 dark:text-destructive/80 shrink-0">
   238→            <AlertCircle className="h-4 w-4" />
   239→          </div>
   240→          <div className="flex-1 min-w-0">
   241→            <p className="text-xs font-semibold">Gagal memuat rekap harian</p>

   243→          </div>
   244→          <Button
   245→            variant="outline"
   246→            size="sm"
   247→            onClick={() => refetch()}
   248→            className="h-7 text-xs px-2 shrink-0"

   250→            <RefreshCw className="h-3 w-3" />

   257→

   261→        <div className="p-4 space-y-3">
   262→          <Skeleton className="h-4 w-24" />
   263→          <Skeleton className="h-10 w-40" />
   264→          <div className="grid grid-cols-3 gap-2">
   265→            <Skeleton className="h-12" />
   266→            <Skeleton className="h-12" />
   267→            <Skeleton className="h-12" />
   268→          </div>
   269→        </div>
   270→      </Card>

   275→
   276→  // ── Empty state: no transactions today ─────────────────────────────
   277→  // Note: budget ring/button is still shown here so the user can set/view

   281→
   282→  if (isEmpty) {

   284→      <Card className="overflow-hidden anim-stagger contain-card">
   285→        <div className="bg-gradient-to-br from-primary/[0.025] via-primary/[0.015] to-transparent px-4 py-4 sm:px-6 sm:py-5">
   286→          <div className="flex items-center justify-between gap-3">

   293→                belum ada transaksi
   294→              </p>
   295→            </div>
   296→            {/* Budget ring (or "Set budget" button) — same as non-empty state */}
   297→            {dailyBudget && dailyBudget.target ? (

   299→                type="button"
   300→                onClick={() => openBudgetDialog(dailyBudget.target)}
   301→                className="flex flex-col items-center gap-1 shrink-0 group cursor-pointer rounded-lg p-1 -m-1 hover:bg-muted/40 transition-colors"
   302→                aria-label={`Budget harian ${dailyBudget.percentage}%, tap to edit`}

   304→                <ProgressRing percentage={dailyBudget.percentage} status={dailyBudget.status} size={48}>

   367→            <div className="flex items-center gap-1.5 mb-1 flex-wrap">
   368→              <p className="text-xs text-muted-foreground font-medium">Hari Ini</p>

   377→              'text-xl sm:text-3xl font-bold tracking-tight break-words',
   378→              today.expense > 0 ? 'text-foreground' : 'text-success'
   379→            )}>
   380→              <CountUpRupiah amount={today.expense} />

   382→            <p className="text-xs text-muted-foreground mt-0.5">
   383→              total pengeluaran · <CountUpNumber value={today.expenseCount} /> transaksi

   385→          </div>
   386→
   387→          {/* Budget ring — tap to edit (custom input dialog).
   388→              If no budget set, show a "Set budget" button instead of the ring. */}
   389→          {dailyBudget && dailyBudget.target ? (
   390→            <button
   391→              type="button"
   392→              onClick={() => openBudgetDialog(dailyBudget.target)}
   393→              className="flex flex-col items-center gap-1 shrink-0 group cursor-pointer rounded-lg p-1 -m-1 hover:bg-muted/40 transition-colors"
   394→              aria-label={`Budget harian ${dailyBudget.percentage}%, tap to edit`}
   395→            >
   396→              <ProgressRing percentage={dailyBudget.percentage} status={dailyBudget.status} size={48}>
   397→                <div className="text-center">
   398→                  <p className="text-[11px] font-bold leading-none">{dailyBudget.percentage}%</p>
   399→                </div>
   400→              </ProgressRing>
   401→              <div className="flex items-center gap-0.5 text-[11px] text-muted-foreground text-center leading-tight">

   407→            <button
   408→              type="button"
   409→              onClick={() => openBudgetDialog(null)}
   410→              className="flex flex-col items-center gap-1 shrink-0 group cursor-pointer rounded-lg p-1.5 -m-1 border border-dashed border-border hover:border-primary/50 hover:bg-primary/5 transition-colors"
   411→              aria-label="Atur budget harian"
   412→            >
   413→              <div className="flex items-center justify-center w-12 h-12 rounded-full text-muted-foreground group-hover:text-primary transition-colors">
   414→                <Target className="h-4 w-4" />
   415→              </div>
   416→              <span className="text-[11px] text-muted-foreground group-hover:text-primary transition-colors">
   417→                Atur budget
   418→              </span>
   419→            </button>
   420→          )}
   421→        </div>
   422→
   423→        {/* Comparison pills row */}
   424→        <div className="flex flex-wrap items-center gap-x-3 gap-y-1.5 mt-2">
   425→          <ComparisonPill
   426→            changePct={comparison.vsYesterday.changePct}
   427→            direction={comparison.vsYesterday.direction}
   428→            label="vs kemarin"
   429→          />
   430→          <ComparisonPill
   431→            changePct={comparison.vs7DayAverage.changePct}
   432→            direction={comparison.vs7DayAverage.direction}

   434→          />
   435→          {predictions.trendDirection.direction !== 'flat' && (
   436→            <div className={cn(
   437→              'flex items-center gap-1 text-xs font-medium min-w-0',

   446→        {/* Sparkline — premium line chart with blue→purple gradient.

   450→          <MiniSparkline data={sparkline.daily7d} />
   451→        </div>
   452→      </div>
   453→
   454→      {/* ── ALERTS (if any) ──────────────────────────────────────────── */}
   455→      {alerts.length > 0 && (
   456→        <>
   457→          <div className="border-t border-border" />
   458→          <div className="px-4 py-2.5 sm:px-6 flex flex-wrap gap-2">
   459→            {alerts.map((alert, i) => (
   460→              <AlertChip key={i} alert={alert} />

   462→          </div>
