    75→    75→ * Signature teal→emerald gradient panel at the very top of the dashboard.
    76→    76→ * The greeting follows the Jakarta wall-clock hour; the date label is
    77→    77→ * formatted in Indonesian from the Jakarta date string (local-midnight
    78→    78→ * Date so the label is stable in any browser timezone). Client-only by
    79→    79→ * design — Dashboard is dynamically imported with ssr:false — but window
    80→    80→ * is still guarded for safety.
    81→    81→ */
    82→    82→function GreetingHero({ successToday }: { successToday: number }) {
    83→    83→  const { greeting, dateLabel } = useMemo(() => {
    84→    84→    if (typeof window === 'undefined') {
    85→    85→      return { greeting: 'Selamat datang 👋', dateLabel: '' };
    86→    86→    }
    87→    87→    const { hours } = jakartaNowParts();
    88→    88→    const greeting =
    89→    89→      hours >= 4 && hours < 11
    90→    90→        ? 'Selamat pagi 🌤'
    91→    91→        : hours >= 11 && hours < 15
    92→    92→          ? 'Selamat siang ☀️'
    93→    93→          : hours >= 15 && hours < 19
    94→    94→            ? 'Selamat sore 🌇'
    95→    95→            : 'Selamat malam 🌙';
    96→    96→    const [y, m, d] = jakartaDateString().split('-').map(Number);
    97→    97→    const dateLabel = new Intl.DateTimeFormat('id-ID', {
    98→    98→      weekday: 'long',
    99→    99→      day: 'numeric',
   100→   100→      month: 'long',
   101→   101→      year: 'numeric',
   102→   102→    }).format(new Date(y, m - 1, d));
   103→   103→    return { greeting, dateLabel };
   104→   104→  }, []);
   105→   105→
   106→   106→  return (
   107→   107→    <div className="premium-hero">
   108→   108→      <div className="premium-hero-bubbles" aria-hidden="true" />
   109→   109→      <div className="relative z-10 flex flex-wrap items-center justify-between gap-x-5 gap-y-3 px-5 py-5 sm:px-6">
   110→   110→        <div className="premium-fade-up min-w-0">
   111→   111→          <h2 className="text-lg font-bold tracking-tight sm:text-xl">{greeting}</h2>
   112→   112→          {dateLabel && <p className="mt-1 text-[13px] font-medium opacity-90">{dateLabel}</p>}
   113→   113→        </div>
   114→   114→        <div
   115→   115→          className="premium-fade-up flex shrink-0 items-center gap-3 rounded-2xl border border-white/25 bg-white/10 px-4 py-2.5"
   116→   116→          style={{ animationDelay: '120ms' }}
   117→   117→        >
   118→   118→          <span className="grid h-9 w-9 shrink-0 place-items-center rounded-xl border border-white/25 bg-white/10">
   119→   119→            <Zap className="h-4.5 w-4.5" aria-hidden="true" />
   120→   120→          </span>
   121→   121→          <span className="block leading-none">
   122→   122→            <span className="premium-stat block text-xl sm:text-2xl">{Math.round(successToday)}%</span>
   123→   123→            <span className="mt-1.5 block text-[10px] font-semibold uppercase tracking-[0.14em] opacity-80">
   124→   124→              Hari ini
   125→   125→            </span>
   126→   126→          </span>
   127→   127→        </div>
   128→   128→      </div>
   129→   129→    </div>
   130→   130→  );
   131→   131→}
