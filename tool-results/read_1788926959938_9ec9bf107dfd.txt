   323→ // ── PHASE4-POLISH: drag-to-reorder (@dnd-kit) ──────────────────────────
   324→ // Sensors + handlers are declared HERE (after `activeHabits` is defined)
   325→ // because handleDragEnd's deps array references `activeHabits` directly —
   326→ // moving them above would hit the temporal-dead-zone on the first render.
   327→ // Sensors: PointerSensor (mouse), TouchSensor (mobile drag — required for

   330→ const sensors = useSensors(

   332→   useSensor(TouchSensor, {
   333→     activationConstraint: { delay: 150, tolerance: 6 },

   336→     coordinateGetter: sortableKeyboardCoordinates,

   340→ const handleDragEnd = useCallback(
   341→   async (event: DragEndEvent) => {
   342→     const { active, over } = event;
   343→     if (!over || active.id === over.id) return;
   344→     const oldIndex = activeHabits.findIndex((h) => h.id === active.id);
   345→     const newIndex = activeHabits.findIndex((h) => h.id === over.id);
   346→     if (oldIndex < 0 || newIndex < 0) return;
   347→
   348→     const reordered = arrayMove(activeHabits, oldIndex, newIndex);
   349→     // Re-assign `order` so the new array position matches the DB order
   350→     // (0..N-1 across active habits). Collect only the diffs to PUT.
   351→     const updates: { id: string; sortOrder: number }[] = [];
   352→     reordered.forEach((h, idx) => {
   353→       if (h.sortOrder !== idx) updates.push({ id: h.id, sortOrder: idx });
   354→     });
   355→
   356→     // Optimistic local override: reordered active habits (with new order
   357→     // field) followed by the unchanged paused/archived habits.
   358→     const activeIds = new Set(reordered.map((h) => h.id));
   359→     const nonActive = habits.filter((h) => !activeIds.has(h.id));
   360→     const reorderedAll: Habit[] = [
   361→       ...reordered.map((h, idx) => ({ ...h, sortOrder: idx })),
   362→       ...nonActive,
   363→     ];
   364→     setLocalHabitsOverride(reorderedAll);
   365→
   366→     // Persist each changed habit's order via PUT /api/habits/[id].
   367→     // Fire-and-forget in parallel; invalidate the query on settle so the
   368→     // server-side truth is re-fetched (and the local override cleared).
   369→     try {
   370→       await Promise.all(
   371→         updates.map((u) =>
   372→           fetch(`/api/habits/${u.id}`, {
   373→             method: 'PUT',
   374→             headers: { 'Content-Type': 'application/json' },
   375→             body: JSON.stringify({ sortOrder: u.sortOrder }),
   376→           }),
   377→         ),
   378→       );
   379→       await queryClient.invalidateQueries({ queryKey: ['habits'] });
   380→       // Brief delay so the refetch lands before we drop the override —
   381→       // otherwise a re-render with the stale query cache could flicker
   382→       // back to the old order for one frame.
   383→       setTimeout(() => setLocalHabitsOverride(null), 200);
   384→     } catch {
   385→       toast.error('Gagal menyimpan urutan');
   386→       // Revert to server truth.
   387→       await queryClient.invalidateQueries({ queryKey: ['habits'] });
   388→       setLocalHabitsOverride(null);
   389→     }
   390→   },
   391→   [activeHabits, habits, queryClient],
   392→ );
   393→

   440→ // Best current streak across all active habits

   444→   if (!cache) return 0;
   445→   let best = 0;

   447→     const logs = cache[h.id] || [];
   448→     // PHASE1-HABIT: pass vacationMode so vacationing habits' streaks don't
   449→     // break during the pause.

   451→     // streak counts consecutive days WITHOUT a relapse.

   454→       invert: h.habitType === 'avoid',
   455→       startDate: h.startDate,
   456→     });

   458→   }

   462→ // ---- fetch completions (month-cached) ----
   463→ // BUGHUNT-ROUND2 RACE-1: `isCancelled` guard (passed by the loading
   464→ // effect) prevents a SLOWER stale fetch (e.g. for the previous date/
   465→ // month after rapid navigation) from overwriting the state of the
   466→ // NEWER fetch that already resolved. Previously only `setLoading` was

   469→ const fetchCompletions = async (
   470→   habitList: Habit[],
   471→   date: string,
   472→   isCancelled?: () => boolean,
   473→ ) => {
   474→   const month = date.slice(0, 7);
   475→
   476→   // BUG-16 fix: include refreshKey in the cache hit check. Without this,
   477→   // changing refreshKey (e.g. after creating a habit) would still hit the
   478→   // stale cache and never re-fetch.
   479→   if (
