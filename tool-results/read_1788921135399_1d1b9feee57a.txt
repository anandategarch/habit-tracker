     4→// Extracted from finance.tsx during SPLIT-PHASE2-UI.
     5→//
     6→// Owns all dialog open-state + form state + CRUD handlers for the Finance
     7→// page (Transaction CRUD, Split helpers, Budget CRUD, Category CRUD, Source
     8→// CRUD, Bulk Delete). The main Finance component consumes this hook's return
     9→// value and passes the relevant slices to the extracted dialog components.
    10→//
    11→// Design notes:
    12→//  - The hook owns the form/dialog state because every handler needs to
    13→//    mutate it (close dialog on success, clear form, etc.). Hoisting state
    14→//    to the parent would require passing ~30 setters back into the hook.
    15→//  - `getActiveSources` is passed in because it depends on the `sources`
    16→//    useQuery in the main component (data-fetching stays there per the
    17→//    split spec).
    18→//  - `invalidateFinance` is created inside the hook (it depends on
    19→//    queryClient + useAppStore.triggerRefresh, both of which the hook can
    20→//    call directly).

   165→     snapshotTimerRef.current = null;

   271→ const handleSubmitTx = useCallback(async (event?: React.MouseEvent<HTMLButtonElement>) => {
   272→   // BUG-5 fix: double-submit guard. If a submission is already in flight
   273→   // (e.g. the user double-clicked Simpan), bail out immediately. The guard
   274→   // is set right before the async fetch and reset in each finally block, so
   275→   // validation-failure early returns below don't leave it stuck as true.
   276→   if (submitGuard.current) return;
   277→   // ── Split mode branch ──
   278→   // Split is only valid for CREATE (not editing — each split child is
   279→   // edited individually via the regular single-category form).
   280→   if (splitMode && !editingTx) {
   281→     // Validate: at least 2 rows with category + amount > 0
   282→     const validRows = splitRows.filter(r => r.category && parseNominalInput(r.amount) && parseInt(parseNominalInput(r.amount), 10) > 0);
   283→     if (validRows.length < 2) { toast.error('Minimal 2 kategori dengan jumlah valid untuk split'); return; }
   284→     if (!txForm.date) { toast.error('Pilih tanggal'); return; }
   285→     if (splitTotal <= 0) { toast.error('Total split harus lebih dari 0'); return; }
   286→     // Parse date+time with explicit Jakarta offset (+07:00) — same
   287→     // pattern as the non-split path below.
   288→     const fullDate = txForm.time
   289→       ? new Date(`${txForm.date}T${txForm.time}:00+07:00`)
   290→       : new Date(`${txForm.date}T00:00:00+07:00`);
   291→     const payload = {
   292→       date: fullDate.toISOString(),
   293→       source: txForm.source,
   294→       description: txForm.description || null,
   295→       splits: validRows.map(r => ({
   296→         category: r.category,
   297→         amount: parseInt(parseNominalInput(r.amount), 10),
   298→       })),
   299→     };
   300→     submitGuard.current = true;
   301→     setSubmitting(true);
   302→     try {
   303→       const res = await fetch('/api/finance/transactions/split', {
   304→         method: 'POST',
   305→         headers: { 'Content-Type': 'application/json' },
   306→         body: JSON.stringify(payload),
   307→       });
   308→       if (res.ok) {
   309→         toast.success(`Split transaction berhasil (${validRows.length} kategori)`);
   310→         setTxDialogOpen(false);
   311→         invalidateFinance();
   312→       } else {
   313→         const err = await res.json().catch(() => null);
   314→         toast.error(err?.error || 'Gagal membuat split transaction');
   315→         return;
   316→       }
   317→     } catch { toast.error('Terjadi kesalahan'); } finally { setSubmitting(false); submitGuard.current = false; }
   318→     return;
   319→   }
   320→
   321→   // ── Regular (non-split) branch ──
   322→   const rawAmount = parseNominalInput(txForm.amount);
   323→   if (!rawAmount || parseFloat(rawAmount) <= 0) { toast.error('Masukkan jumlah yang valid'); return; }
   324→   if (!txForm.category) { toast.error('Pilih kategori'); return; }
   325→   if (!txForm.date) { toast.error('Pilih tanggal'); return; }
   326→   // Audit fix: parse date+time with explicit Jakarta offset (+07:00).
   327→   // Previously `new Date('2025-01-15T14:30:00')` (no offset) parsed as
   328→   // the BROWSER's local TZ — so a user in Tokyo (UTC+9) submitting "14:30"
   329→   // would store an epoch that corresponds to 12:30 Jakarta. Now the
   330→   // epoch always corresponds to the Jakarta wall-clock the user typed,
   331→   // regardless of their device timezone. Consistent with openNewTx/openEditTx
   332→   // which both prefill using Jakarta TZ.
   333→   const fullDate = txForm.time
   334→     ? new Date(`${txForm.date}T${txForm.time}:00+07:00`)
   335→     : new Date(`${txForm.date}T00:00:00+07:00`);
   336→   const payload = { ...txForm, amount: rawAmount, date: fullDate.toISOString() };
   337→   submitGuard.current = true;
   338→   setSubmitting(true);
   339→   try {
   340→     if (editingTx) {
   341→       const res = await fetch(`/api/finance/transactions/${editingTx.id}`, { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) });
   342→       // BUGHUNT-ROUND2 TOAST-ERR: surface the API's specific error message
   343→       // (e.g. "Saldo sumber dana tidak mencukupi...") instead of a generic
   344→       // one — matches the delete handler's behavior.
   345→       if (res.ok) toast.success('Transaksi berhasil diupdate');
   346→       else {
   347→         const err = await res.json().catch(() => null);
   348→         toast.error(err?.error || 'Gagal mengupdate transaksi');
   349→         return;
   350→       }
   351→     } else {
   352→       const res = await fetch('/api/finance/transactions', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) });
   353→       if (res.ok) { toast.success('Transaksi berhasil ditambahkan'); smallPop(event?.currentTarget); } else { toast.error('Gagal menambahkan transaksi'); return; }
   354→       // ANIM-2 / Feature 3: fire money particles when an income transaction
   355→       // is added. Split mode is expense-only, so this only applies to the
   356→       // regular single-transaction branch (txForm.type === 'income').
   357→       if (txForm.type === 'income') {
   358→         setMoneyParticlesKey(k => k + 1);
   359→       }
   360→     }
   361→     setTxDialogOpen(false);
   362→     invalidateFinance();
   363→   } catch { toast.error('Terjadi kesalahan'); } finally { setSubmitting(false); submitGuard.current = false; }
   364→ }, [splitMode, editingTx, splitRows, txForm, splitTotal, invalidateFinance]);

   441→   setSubmitting(true);

   491→ const handleSaveBalance = useCallback(async (sourceId: string) => {
   492→   const raw = parseNominalInput(balanceEditValue);
   493→   const val = parseFloat(raw);
   494→   if (isNaN(val)) { setBalanceEditId(null); setBalanceEditValue(''); return; }
   495→
   496→   // Bug fix: skip PATCH if value unchanged (user clicked then blurred
   497→   // without editing). Avoids unnecessary network call + toast spam.
   498→   const source = getActiveSources().find((s) => s.id === sourceId);
   499→   const currentBalance = source?.balance ?? 0;
   500→   if (val === currentBalance) {
   501→     setBalanceEditId(null);
   502→     setBalanceEditValue('');
   503→     return;
   504→   }
   505→
   506→   try {
   507→     const res = await fetch(`/api/finance/sources/${sourceId}/balance`, { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ balance: val }) });
   508→     if (res.ok) {
   509→       const data = await res.json();
   510→       // Invalidate both sources + transactions (adjustment creates a tx)
   511→       queryClient.invalidateQueries({ queryKey: ['finance', 'sources'] });
   512→       queryClient.invalidateQueries({ queryKey: ['finance', 'transactions'] });
   513→       queryClient.invalidateQueries({ queryKey: ['finance', 'daily-recap'] });
   514→       // Show informative toast based on whether an adjustment was made
   515→       if (data?.adjustment) {
   516→         const adj = data.adjustment;
   517→         const sign = adj.type === 'income' ? '+' : '−';
   518→         toast.success(`Saldo diupdate — transaksi "${sign}${formatRupiah(adj.amount)}" dibuat`);
   519→       } else {
   520→         toast.success('Saldo diupdate');
   521→       }
   522→     } else {
   523→       toast.error('Gagal update saldo');
   524→     }
   525→   } catch { toast.error('Gagal update saldo'); }
   526→   setBalanceEditId(null); setBalanceEditValue('');
   527→ }, [balanceEditValue, getActiveSources, queryClient]);
