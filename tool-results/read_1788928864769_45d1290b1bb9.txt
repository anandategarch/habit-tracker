   427→const DOCK_SIDE = 0.43; // width share of each tab group (left/right)

   430→const FAB_PROTRUDE = 22; // px of FAB protruding above the dock top edge

   432→const NAV_LEFT_ITEMS: { id: TabId; label: string; icon: React.ElementType }[] = [
   433→ { id: 'dashboard', label: 'Beranda', icon: LayoutDashboard },

   438→const NAV_RIGHT_ITEMS: { id: TabId; label: string; icon: React.ElementType }[] = [

   443→// Geometri indikator per tab (fraksi lebar dock) — data-driven dari daftar
   444→// di atas, jadi menambah/menggeser tab tidak perlu menyentuh konstanta
   445→// posisi manual. Setiap tab punya {left, width} dalam fraksi dockW.
   446→function dockTabMetrics(id: TabId): { left: number; width: number } | undefined {
   447→ const li = NAV_LEFT_ITEMS.findIndex((n) => n.id === id);
   448→ if (li >= 0) {
   449→   const w = DOCK_SIDE / NAV_LEFT_ITEMS.length;
   450→   return { left: w * li, width: w };
   451→ }
   452→ const ri = NAV_RIGHT_ITEMS.findIndex((n) => n.id === id);
   453→ if (ri >= 0) {
   454→   const w = DOCK_SIDE / NAV_RIGHT_ITEMS.length;
   455→   return { left: 1 - DOCK_SIDE + w * ri, width: w };
   456→ }
   457→ return undefined;
   458→}
