   280→   280→              method: 'PUT',

   397→   397→      setCompletedAtMap(atMap);
   398→   398→      return;
   399→   399→    }
   400→   400→
   401→   401→    const active = habitList.filter((h) => h.status === 'active');
   402→   402→    const ids = active.map((h) => h.id);
   403→   403→
   404→   404→    let groupedLogs: Record<string, HabitLog[]> = {};
   405→   405→    try {
   406→   406→      const res = await fetch(
   407→   407→        `/api/habits/batch-logs?month=${month}&habitIds=${ids.join(',')}`,
   408→   408→      );
   409→   409→      if (res.ok) {
   410→   410→        groupedLogs = await res.json();
   411→   411→      }
   412→   412→    } catch {
   413→   413→      // fall through to empty defaults
   414→   414→    }
   415→   415→
   416→   416→    // RACE-1: a newer fetch may have resolved while this one was in flight.
   417→   417→    // Bailing here avoids: (a) painting stale completion maps over the
   418→   418→    // current date's state, (b) flipping cachedMonthRef back to the stale
   419→   419→    // month (which would only cost a redundant re-fetch later, but still).
   420→   420→    if (isCancelled?.()) return;
