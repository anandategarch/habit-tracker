    74→/* ── PREMIUM UI v2 ("Rutina Aurora") — Hero greeting ─────────────────────
    75→ * Signature teal→emerald gradient panel at the very top of the dashboard.
    76→ * The greeting follows the Jakarta wall-clock hour; the date label is
    77→ * formatted in Indonesian from the Jakarta date string (local-midnight
    78→ * Date so the label is stable in any browser timezone). Client-only by
    79→ * design — Dashboard is dynamically imported with ssr:false — but window
    80→ * is still guarded for safety.
    81→ */
    82→function GreetingHero({ successToday }: { successToday: number }) {
    83→  const { greeting, dateLabel } = useMemo(() => {
    84→    if (typeof window === 'undefined') {
    85→      return { greeting: 'Selamat datang 👋', dateLabel: '' };
    86→    }
    87→    const { hours } = jakartaNowParts();
    88→    const greeting =
    89→      hours >= 4 && hours < 11
    90→        ? 'Selamat pagi 🌤'
    91→        : hours >= 11 && hours < 15
    92→          ? 'Selamat siang ☀️'
    93→          : hours >= 15 && hours < 19
    94→            ? 'Selamat sore 🌇'
    95→            : 'Selamat malam 🌙';
    96→    const [y, m, d] = jakartaDateString().split('-').map(Number);
    97→    const dateLabel = new Intl.DateTimeFormat('id-ID', {
    98→      weekday: 'long',
    99→      day: 'numeric',
   100→      month: 'long',
   101→      year: 'numeric',
   102→    }).format(new Date(y, m - 1, d));
   103→    return { greeting, dateLabel };
   104→  }, []);
   105→
   106→  return (
   107→    <div className="premium-hero">
   108→      <div className="premium-hero-bubbles" aria-hidden="true" />
   109→      <div className="relative z-10 flex flex-wrap items-center justify-between gap-x-5 gap-y-3 px-5 py-5 sm:px-6">
   110→        <div className="premium-fade-up min-w-0">
   111→          <h2 className="text-lg font-bold tracking-tight sm:text-xl">{greeting}</h2>
   112→          {dateLabel && <p className="mt-1 text-[13px] font-medium opacity-90">{dateLabel}</p>}
   113→        </div>
   114→        <div
   115→          className="premium-fade-up flex shrink-0 items-center gap-3 rounded-2xl border border-white/25 bg-white/10 px-4 py-2.5"
   116→          style={{ animationDelay: '120ms' }}
   117→        >
   118→          <span className="grid h-9 w-9 shrink-0 place-items-center rounded-xl border border-white/25 bg-white/10">
   119→            <Zap className="h-4.5 w-4.5" aria-hidden="true" />
   120→          </span>
   121→          <span className="block leading-none">
   122→            <span className="premium-stat block text-xl sm:text-2xl">{Math.round(successToday)}%</span>
   123→            <span className="mt-1.5 block text-[10px] font-semibold uppercase tracking-[0.14em] opacity-80">
   124→              Hari ini
   125→            </span>
   126→          </span>
   127→        </div>
   128→      </div>
   129→    </div>
   130→  );
   131→}
