    66→/** Live preview theme colors without saving to DB */
    67→function previewTheme(primary: string, secondary: string, theme: string) {
    68→  const isDark = theme === 'dark';
    69→  applyThemeMode(isDark);
    70→
    71→  if (
    72→    primary.toLowerCase() === '#22c55e' &&
    73→    secondary.toLowerCase() === '#10b981'
    74→  ) {
    75→    resetThemeColors();
    76→  } else {
    77→    applyThemeColors(primary, secondary, isDark);
    78→  }
    79→}
