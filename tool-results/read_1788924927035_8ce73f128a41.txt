    92→ const saveProjectionCategoriesMutation = useMutation({
    93→   mutationFn: async (categoryNames: string[]) => {
    94→     const res = await fetch('/api/settings', {
    95→       method: 'PUT',
    96→       headers: { 'Content-Type': 'application/json' },
    97→       body: JSON.stringify({ projectionCategoryIds: categoryNames }),
    98→     });
    99→     if (!res.ok) throw new Error('Failed to save projection categories');
   100→     return res.json();
   101→   },
   102→   onMutate: async (categoryNames: string[]) => {
   103→     // Cancel any outgoing refetches so they don't overwrite our optimistic
   104→     // update.
   105→     await queryClient.cancelQueries({ queryKey: ['finance', 'daily-recap'] });
   106→     // Snapshot the previous value for rollback.
   107→     const previousRecap = queryClient.getQueryData<DailyRecapData>(['finance', 'daily-recap']);
   108→     // Optimistically patch the cached recap's projectionCategoryNames +
   109→     // projectionIsFiltered so the dropdown updates instantly.
   110→     //
   111→     // BUG-3 fix: Also reset whatIfBase to 0. The what-if slider computes
   112→     // its result from `whatIfBase` (month-to-date spend of the CURRENT
   113→     // basis). Without resetting it, the slider would use the STALE
   114→     // whatIfBase (from the old filter) for 300-800ms until the refetch
   115→     // completes — producing wrong numbers. Setting whatIfBase to 0
   116→     // hides the slider via its `whatIfBase > 0` guard, so the user sees
   117→     // a clean "loading" state instead of wrong numbers. The slider
   118→     // reappears with the correct whatIfBase once the refetch completes.
   119→     if (previousRecap) {
   120→       queryClient.setQueryData<DailyRecapData>(['finance', 'daily-recap'], {
   121→         ...previousRecap,
   122→         predictions: {
   123→           ...previousRecap.predictions,
   124→           projectionCategoryNames: categoryNames,
   125→           projectionIsFiltered: categoryNames.length > 0,
   126→           whatIfBase: 0,
   127→         },
   128→       });
   129→     }
   130→     return { previousRecap };
   131→   },
   132→   onSuccess: () => {
   133→     queryClient.invalidateQueries({ queryKey: ['finance', 'daily-recap'] });
   134→     queryClient.invalidateQueries({ queryKey: ['settings'] });
   135→   },

   138→     if (context?.previousRecap) {
