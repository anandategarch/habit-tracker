   335→   335→ const handleDragEnd = useCallback(
   336→   336→   async (event: DragEndEvent) => {
   337→   337→     const { active, over } = event;
   338→   338→     if (!over || active.id === over.id) return;
   339→   339→     const oldIndex = activeHabits.findIndex((h) => h.id === active.id);
   340→   340→     const newIndex = activeHabits.findIndex((h) => h.id === over.id);
   341→   341→     if (oldIndex < 0 || newIndex < 0) return;
   342→   342→
   343→   343→     const reordered = arrayMove(activeHabits, oldIndex, newIndex);
   344→   344→     // Re-assign `order` so the new array position matches the DB order
   345→   345→     // (0..N-1 across active habits). Collect only the diffs to PUT.
   346→   346→     const updates: { id: string; sortOrder: number }[] = [];
   347→   347→     reordered.forEach((h, idx) => {
   348→   348→       if (h.sortOrder !== idx) updates.push({ id: h.id, sortOrder: idx });
   349→   349→     });
   350→   350→
   351→   351→     // Optimistic local override: reordered active habits (with new order
   352→   352→     // field) followed by the unchanged paused/archived habits.
   353→   353→     const activeIds = new Set(reordered.map((h) => h.id));
   354→   354→     const nonActive = habits.filter((h) => !activeIds.has(h.id));
   355→   355→     const reorderedAll: Habit[] = [
   356→   356→       ...reordered.map((h, idx) => ({ ...h, sortOrder: idx })),
   357→   357→       ...nonActive,
   358→   358→     ];
   359→   359→     setLocalHabitsOverride(reorderedAll);
   360→   360→
   361→   361→     // Persist each changed habit's order via PUT /api/habits/[id].
   362→   362→     // Fire-and-forget in parallel; invalidate the query on settle so the
   363→   363→     // server-side truth is re-fetched (and the local override cleared).
   364→   364→     try {
   365→   365→       await Promise.all(
   366→   366→         updates.map((u) =>
   367→   367→           fetch(`/api/habits/${u.id}`, {
   368→   368→             method: 'PUT',
   369→   369→             headers: { 'Content-Type': 'application/json' },
   370→   370→             body: JSON.stringify({ sortOrder: u.sortOrder }),
   371→   371→           }),
   372→   372→         ),
   373→   373→       );
   374→   374→       await queryClient.invalidateQueries({ queryKey: ['habits'] });
   375→   375→       // Brief delay so the refetch lands before we drop the override —
   376→   376→       // otherwise a re-render with the stale query cache could flicker
   377→   377→       // back to the old order for one frame.
   378→   378→       setTimeout(() => setLocalHabitsOverride(null), 200);
   379→   379→     } catch {
   380→   380→       toast.error('Gagal menyimpan urutan');
   381→   381→       // Revert to server truth.
   382→   382→       await queryClient.invalidateQueries({ queryKey: ['habits'] });
   383→   383→       setLocalHabitsOverride(null);
   384→   384→     }
   385→   385→   },
   386→   386→   [activeHabits, habits, queryClient],
   387→   387→ );
   388→   388→
   389→   389→ /** Toggle drag mode. When enabling, force viewFilter to "all" so every
   390→   390→  *  active habit is visible for reordering. When disabling, drop the

   403→   403→ // (i.e. completionMap[h.id] is false). For "normal" + "amount" habits,
   404→   404→ // success = completionMap[h.id] is true. This derived flag drives the
   405→   405→ // daily summary's completedCount / completionPct / todayXP / bestStreak
   406→   406→ // so an avoid habit that wasn't checked today counts as a success.
