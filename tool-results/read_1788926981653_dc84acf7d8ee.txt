    63→// ── Types ──────────────────────────────────────────────────────────────────
    64→interface DayData {
    65→  date: Date; // UTC-midnight (konvensi date-utils)
    66→  dayStr: string; // yyyy-MM-dd
    67→  dayNum: number;
    68→  isCurrentMonth: boolean;

    76→// ── Helpers ────────────────────────────────────────────────────────────────
    77→// Urutan weekday mengikuti weekStart pengguna (AppSettings.weekStart: 0 =
    78→// Minggu, 1 = Senin — default Senin).
    79→const WEEKDAYS_BASE = ['Min', 'Sen', 'Sel', 'Rab', 'Kam', 'Jum', 'Sab'];
    80→
    81→function weekStartsOnNum(weekStart: number | string | null | undefined): 0 | 1 {
    82→  if (weekStart === 0 || weekStart === 'sunday') return 0;
    83→  return 1; // Senin (default UI Pengaturan)
    84→}
    85→
    86→function rotateWeekdays(weekStartsOn: 0 | 1): string[] {
    87→  return [...WEEKDAYS_BASE.slice(weekStartsOn), ...WEEKDAYS_BASE.slice(0, weekStartsOn)];
    88→}
    89→
    90→// Heatmap Aurora teal berjenjang. 0% = tint destructive (hari terlacak tapi
    91→// kosong = status, bukan level panas); sel tinggi butuh teks putih (AA).
    92→function getHeatmapColor(rate: number | null): string {
    93→  if (rate === null) return 'bg-muted/60';
    94→  if (rate === 0) return 'bg-destructive/25 dark:bg-destructive/15';
    95→  if (rate < 50) return 'bg-teal-200/70 dark:bg-teal-900/50';
    96→  if (rate < 75) return 'bg-teal-400/80 dark:bg-teal-700/60';
    97→  return 'bg-teal-600 dark:bg-teal-500';
    98→}
    99→
   100→function getHeatmapTextColor(rate: number | null): string {
   101→  if (rate === null) return 'text-muted-foreground';
   102→  if (rate === 0) return 'text-destructive dark:text-destructive/80';
   103→  if (rate >= 75) return 'text-white';
   104→  return 'text-foreground';
   105→}
   106→
   107→function getDayNumTextColor(day: DayData): string {
   108→  if (day.completionRate !== null && day.completionRate >= 75) return 'text-white font-bold';
   109→  if (day.isToday) return 'text-primary font-bold';
   110→  if (!day.isCurrentMonth) return 'text-muted-foreground';
   111→  return 'text-foreground';
   112→}
   113→
   114→function getHeatmapHover(rate: number | null): string {
   115→  if (rate === null || rate === 0) return 'hover:bg-accent/60';
   116→  return 'hover:brightness-105 hover:ring-1 hover:ring-ring/60';
   117→}
   118→
   119→// Label aria Indonesia, mis. "Rabu 15 Januari 2025, 3 dari 5 habit selesai".
   120→function buildDayAriaLabel(day: DayData): string {
   121→  const d = day.date;
   122→  const dateLabel = `${eeeeIdFormatter(d)} ${d.getUTCDate()} ${format(d, 'MMMM')} ${format(d, 'yyyy')}`;
   123→  if (!day.isCurrentMonth) return dateLabel;
   124→  if (day.completionRate === null) return `${dateLabel}, belum ada data`;
   125→  if (day.totalHabits > 0) {
   126→    return `${dateLabel}, ${day.completedHabits} dari ${day.totalHabits} habit selesai`;
   127→  }
   128→  return `${dateLabel}, semua catatan selesai`;
   129→}
   130→
   131→const pad2 = (n: number) => (n < 10 ? `0${n}` : String(n));
   132→
   133→function generateMonthOptions(): { value: string; label: string }[] {
   134→  // Basis bulan = komponen lokal browser (grid kalender memang lokal); label
   135→  // dibangun dari Date.UTC supaya format() tidak meleset di TZ non-UTC.
   136→  const now = new Date();
   137→  const y0 = now.getFullYear();
   138→  const m0 = now.getMonth();
   139→  const options: { value: string; label: string }[] = [];
   140→  for (let i = -12; i <= 3; i++) {
   141→    const total = m0 + i;
   142→    const y = y0 + Math.floor(total / 12);
   143→    const m = ((total % 12) + 12) % 12;
   144→    const d = new Date(Date.UTC(y, m, 1));
   145→    options.push({
   146→      value: `${y}-${pad2(m + 1)}`,
   147→      label: format(d, 'MMMM yyyy'),
   148→    });
   149→  }
   150→  return options;
   151→}

   162→// ── Component ──────────────────────────────────────────────────────────────
   163→export default function CalendarView() {
   164→  const selectedMonth = useAppStore((s) => s.selectedMonth);
   165→  const setSelectedMonth = useAppStore((s) => s.setSelectedMonth);
   166→  // Day-cell tap → tracker grid dengan tanggal terpilih (1-klik).
   167→  const openTrackerDate = useAppStore((s) => s.openTrackerDate);
   168→  const queryClient = useQueryClient();
   169→
   170→  // ── AppSettings (weekStart) — share cache ['settings'] dengan Settings ──
   171→  const { data: settings = null } = useQuery<AppSettings | null>({
   172→    queryKey: ['settings'],
   173→    queryFn: async () => {
   174→      const res = await fetch('/api/settings');
   175→      if (!res.ok) return null;
   176→      return res.json();
   177→    },
   178→    staleTime: 60_000,
   179→  });
   180→  const weekStartsOn = weekStartsOnNum(settings?.weekStart);
   181→  const WEEKDAYS = useMemo(() => rotateWeekdays(weekStartsOn), [weekStartsOn]);
   182→
   183→  const monthOptions = useMemo(() => generateMonthOptions(), []);
   184→
   185→  // ── Bulan berjalan — Date.UTC (konvensi komponen UTC lib/date-utils) ──
   186→  const monthDate = useMemo(() => {
   187→    const [y, m] = selectedMonth.split('-').map(Number);
   188→    return new Date(Date.UTC(y, m - 1, 1));
   189→  }, [selectedMonth]);
   190→
   191→  const monthLabel = useMemo(() => format(monthDate, 'MMMM yyyy'), [monthDate]);
   192→
   193→  // ── Fetch data ────────────────────────────────────────────────────────
   194→  const { data: habits = [], isLoading: habitsLoading } = useQuery<Habit[]>({
   195→    queryKey: ['habits'],
   196→    queryFn: async () => {
   197→      const res = await fetch('/api/habits');
   198→      if (!res.ok) return [];
   199→      const json = (await res.json()) as { habits?: Habit[] };
   200→      return json.habits ?? [];
   201→    },
   202→    staleTime: 30_000,
   203→  });
   204→
   205→  const { data: dailyLogs = [] } = useQuery<DailyLog[]>({
   206→    queryKey: ['daily-logs-month', selectedMonth],
   207→    queryFn: async () => {
   208→      const res = await fetch(`/api/daily-logs?month=${selectedMonth}`);
   209→      if (!res.ok) return [];
   210→      const json = (await res.json()) as { logs?: DailyLog[] };
   211→      return json.logs ?? [];
   212→    },
   213→    staleTime: 30_000,
   214→  });
   215→
   216→  const habitIds = useMemo(() => habits.map((h) => h.id).join(','), [habits]);
   217→  const { data: habitLogs = [], isError: fetchError } = useQuery<HabitLog[]>({
   218→    queryKey: ['habit-logs-batch', selectedMonth, habitIds],
   219→    queryFn: async () => {
   220→      if (!habitIds) return [];
   221→      const res = await fetch(
   222→        `/api/habits/batch-logs?month=${selectedMonth}&ids=${habitIds}`,
   223→      );
   224→      if (!res.ok) return [];
   225→      const json = (await res.json()) as unknown;
   226→      // Kontrak: { logs: HabitLog[] } (flat). Toleransi bentuk grouped lama.
   227→      if (Array.isArray(json)) return json as HabitLog[];
   228→      if (json && typeof json === 'object' && Array.isArray((json as { logs?: HabitLog[] }).logs)) {
   229→        return (json as { logs: HabitLog[] }).logs;
   230→      }
   231→      return Object.values(json as Record<string, HabitLog[]>).flat();
   232→    },
   233→    enabled: habits.length > 0,
   234→    staleTime: 30_000,
   235→  });
   236→
   237→  const loading = habitsLoading;
   238→
   239→  // ── Lookup daily-log per YMD ──────────────────────────────────────────
   240→  const dailyLogMap = useMemo(() => {
   241→    const map: Record<string, DailyLog> = {};
   242→    dailyLogs.forEach((d) => {
   243→      const key = d.date?.slice(0, 10);
   244→      if (key) map[key] = d;
   245→    });
   246→    return map;
   247→  }, [dailyLogs]);
   248→
   249→  // ── Lookup log habit per YMD ──────────────────────────────────────────
   250→  const habitLogMap = useMemo(() => {
   251→    const map: Record<string, { completed: number; total: number }> = {};
   252→    habitLogs.forEach((log) => {
   253→      const key = log.date?.slice(0, 10);
   254→      if (!key) return;
   255→      if (!map[key]) map[key] = { completed: 0, total: 0 };
   256→      map[key].total++;
   257→      if (log.completed) map[key].completed++;
   258→    });
   259→    return map;
   260→  }, [habitLogs]);
   261→
   262→  // ── Sel kalender ──────────────────────────────────────────────────────
   263→  const calendarDays = useMemo<DayData[]>(() => {
   264→    const monthStart = startOfMonth(monthDate);
   265→    const monthEnd = endOfMonth(monthDate);
   266→    const calStart = startOfWeek(monthStart, { weekStartsOn });
   267→    const calEnd = endOfWeek(monthEnd, { weekStartsOn });
   268→    const days = eachDayOfInterval(calStart, calEnd);
   269→
   270→    const today = new Date();
   271→    // Habit aktif pada hari tsb: tidak diarsipkan & sudah mulai (perbandingan
   272→    // YMD string — TZ-safe, tanpa parse lokal).
   273→    const activeHabitCountOnDay = (dayStr: string): number =>
   274→      habits.filter((h) => {
   275→        if (h.isArchived) return false;
   276→        const start = h.startDate?.slice(0, 10);
   277→        if (start && start > dayStr) return false;
   278→        return true;
   279→      }).length;
   280→
   281→    return days.map((d) => {
   282→      const dayStr = format(d, 'yyyy-MM-dd');
   283→      const hLogs = habitLogMap[dayStr];
   284→      const dLog = dailyLogMap[dayStr];
   285→      const inMonth = isSameMonth(d, monthDate);
   286→      const isFuture = isBefore(today, startOfDay(d)) && !isToday(d);
   287→
   288→      const totalHabitsOnDay = inMonth && !isFuture ? activeHabitCountOnDay(dayStr) : 0;
   289→
   290→      let completionRate: number | null = null;
   291→      if (!inMonth || isFuture) {
   292→        completionRate = null;
   293→      } else if (hLogs && hLogs.total > 0 && totalHabitsOnDay > 0) {
   294→        completionRate = Math.round((hLogs.completed / totalHabitsOnDay) * 100);
   295→      } else if (hLogs && hLogs.total > 0 && totalHabitsOnDay === 0) {
   296→        completionRate = 100;
   297→      } else {
   298→        completionRate = null;
   299→      }
   300→
   301→      return {
   302→        date: d,
   303→        dayStr,
   304→        dayNum: d.getUTCDate(),
   305→        isCurrentMonth: inMonth,
   306→        isToday: isToday(d),
   307→        completionRate,
   308→        mood: dLog?.mood ?? null,
   309→        totalHabits: totalHabitsOnDay,
   310→        completedHabits: hLogs?.completed ?? 0,
   311→      };
   312→    });
   313→  }, [monthDate, habits, habitLogMap, dailyLogMap, weekStartsOn]);
   314→
   315→  // ── Ringkasan bulan ───────────────────────────────────────────────────
   316→  const monthSummary = useMemo(() => {
   317→    const activeDays = calendarDays.filter(
   318→      (d) => d.isCurrentMonth && d.completionRate !== null,
   319→    );
   320→    if (activeDays.length === 0) return { avg: 0, best: null, worst: null, entries: 0 };
   321→
   322→    const rates = activeDays.map((d) => (d.completionRate as number) || 0);
   323→    const avg = rates.reduce((a, b) => a + b, 0) / rates.length;
   324→    const bestIdx = rates.indexOf(Math.max(...rates));
   325→    const worstIdx = rates.indexOf(Math.min(...rates));
   326→
   327→    return {
   328→      avg,
   329→      best: activeDays[bestIdx],
   330→      worst: activeDays[worstIdx],
   331→      entries: activeDays.length,
   332→    };
   333→  }, [calendarDays]);
   334→
   335→  // ── Navigasi ──────────────────────────────────────────────────────────
   336→  const goToPrevMonth = useCallback(() => {
   337→    const prev = subMonths(monthDate, 1);
   338→    setSelectedMonth(format(prev, 'yyyy-MM'));
   339→  }, [monthDate, setSelectedMonth]);
   340→
   341→  const goToNextMonth = useCallback(() => {
   342→    const next = addMonths(monthDate, 1);
   343→    setSelectedMonth(format(next, 'yyyy-MM'));
   344→  }, [monthDate, setSelectedMonth]);
   345→
   346→  const handleDayClick = useCallback(
   347→    (dayStr: string) => openTrackerDate(dayStr),
   348→    [openTrackerDate],
   349→  );
   350→
   351→  const retryFetch = useCallback(() => {
   352→    queryClient.invalidateQueries({ queryKey: ['habits'] });
   353→    queryClient.invalidateQueries({ queryKey: ['daily-logs-month'] });
   354→    queryClient.invalidateQueries({ queryKey: ['habit-logs-batch'] });
   355→  }, [queryClient]);
   356→
   357→  // ── Render ────────────────────────────────────────────────────────────
   358→  return (
   359→    <div className="space-y-6">
   360→      {/* ── Header + navigasi bulan ── */}
   361→      <PageHeader
   362→        title="Kalender"
   363→        subtitle="Visualisasikan penyelesaian habit kamu dalam heatmap bulanan."
   364→      >
   365→        <Button
   366→          variant="outline"
   367→          size="icon"
   368→          onClick={goToPrevMonth}
   369→          aria-label="Bulan sebelumnya"
   370→        >
   371→          <ChevronLeft className="h-4 w-4" />
   372→        </Button>
   373→
   374→        <Select value={selectedMonth} onValueChange={setSelectedMonth}>
   375→          <SelectTrigger className="w-[168px]" aria-label="Pilih bulan">
   376→            <SelectValue />
   377→          </SelectTrigger>
   378→          <SelectContent>
   379→            {monthOptions.map((opt) => (
   380→              <SelectItem key={opt.value} value={opt.value}>
   381→                {opt.label}
   382→              </SelectItem>
   383→            ))}
   384→          </SelectContent>
   385→        </Select>
   386→
   387→        <Button
   388→          variant="outline"
   389→          size="icon"
   390→          onClick={goToNextMonth}
   391→          aria-label="Bulan berikutnya"
   392→        >
   393→          <ChevronRight className="h-4 w-4" />
   394→        </Button>
   395→      </PageHeader>
   396→
   397→      {fetchError && (
   398→        <div className="premium-card premium-empty rounded-2xl">
   399→          <div className="premium-empty-orb" aria-hidden="true">
   400→            <CalendarDays className="h-8 w-8 text-primary" />
   401→          </div>
   402→          <p className="text-sm font-medium text-foreground">
   403→            Gagal memuat data kalender
   404→          </p>
   405→          <p className="text-xs text-muted-foreground/70 -mt-0.5">
   406→            Coba muat ulang habit dan log kamu.
   407→          </p>
   408→          <Button
   409→            size="sm"
   410→            className="btn-primary-gradient anim-press"
   411→            onClick={retryFetch}
   412→          >
   413→            <RefreshCw className="h-4 w-4" />
   414→            Coba Lagi
   415→          </Button>
   416→        </div>
   417→      )}
   418→
   419→      {!fetchError &&
   420→        (loading ? (
   421→          <div
   422→            className="premium-card premium-card-sheen rounded-2xl p-4 sm:p-5"
   423→            aria-busy="true"
   424→            aria-label="Memuat kalender"
   425→          >
   426→            <div className="grid grid-cols-7 gap-1.5 mb-3">
   427→              {WEEKDAYS.map((d) => (
   428→                <Skeleton key={d} className="h-4 rounded-md" />
   429→              ))}
   430→            </div>
   431→            <div className="grid grid-cols-7 gap-1.5 max-h-[420px] overflow-hidden">
   432→              {Array.from({ length: 35 }).map((_, i) => (
   433→                <Skeleton
   434→                  key={i}
   435→                  className="h-[60px] sm:h-[68px] rounded-lg"
   436→                  style={{ animationDelay: `${(i % 7) * 70}ms` }}
   437→                />
   438→              ))}
   439→            </div>
   440→          </div>
   441→        ) : (
   442→          <>
   443→            {/* ── Grid kalender (div premium-card, bukan Card shadcn) ── */}
   444→            <div className="premium-card premium-card-sheen rounded-2xl p-4 sm:p-5 premium-fade-up">
   445→              {/* Header weekday */}
   446→              <div className="grid grid-cols-7 gap-1 mb-2">
   447→                {WEEKDAYS.map((d) => (
   448→                  <div
   449→                    key={d}
   450→                    className="text-center text-xs font-semibold text-muted-foreground py-2"
   451→                  >
   452→                    {d}
   453→                  </div>
   454→                ))}
   455→              </div>
   456→
   457→              {/* Day cells — setiap hari tombol nyata → openTrackerDate */}
   458→              <div className="grid grid-cols-7 gap-1">
   459→                {calendarDays.map((day) => (
   460→                  <button
   461→                    key={day.dayStr}
   462→                    type="button"
   463→                    onClick={() => handleDayClick(day.dayStr)}
   464→                    aria-label={buildDayAriaLabel(day)}
   465→                    className={`
   466→                      relative min-h-[72px] sm:min-h-[88px] md:min-h-[100px] rounded-lg p-1.5 sm:p-2 text-left
   467→                      transition-all duration-150 cursor-pointer active:scale-95
   468→                      focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring/60
   469→                      ${getHeatmapColor(day.completionRate)}
   470→                      ${getHeatmapHover(day.completionRate)}
   471→                      ${!day.isCurrentMonth ? 'opacity-35' : ''}
   472→                      ${day.isToday ? 'ring-2 ring-primary ring-offset-2 ring-offset-background anim-glow-breathe' : ''}
   473→                    `}
   474→                  >
   475→                    <div className="flex items-center justify-between">
   476→                      <span
   477→                        className={`text-xs sm:text-sm font-medium ${getDayNumTextColor(day)}`}
   478→                      >
   479→                        {day.dayNum}
   480→                      </span>
   481→                      {/* Mood marker dari daily-logs */}
   482→                      {day.isCurrentMonth && day.mood !== null && (
   483→                        <span className="text-[10px] leading-none" aria-hidden="true">
   484→                          {MOOD_EMOJIS[Math.round(day.mood)] ?? '🙂'}
   485→                        </span>
   486→                      )}
   487→                    </div>
   488→
   489→                    {day.isCurrentMonth && day.completionRate !== null && (
   490→                      <div className="mt-1 flex flex-col items-center gap-0.5">
   491→                        <span
   492→                          className={`text-xs font-bold ${getHeatmapTextColor(day.completionRate)}`}
   493→                        >
   494→                          {day.completionRate}%
   495→                        </span>
   496→                        <div className="w-full bg-foreground/10 rounded-full h-1 overflow-hidden">
   497→                          <div
   498→                            className="h-full rounded-full bg-foreground/60 transition-all duration-300"
   499→                            style={{
   500→                              width: `${Math.max(day.completionRate, 0)}%`,
   501→                            }}
   502→                          />
   503→                        </div>
   504→                      </div>
   505→                    )}
   506→
   507→                    {day.isCurrentMonth &&
   508→                      day.completionRate === null &&
   509→                      !isBefore(new Date(), startOfDay(day.date)) && (
