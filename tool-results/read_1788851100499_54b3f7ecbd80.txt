    99→// ── Helpers ────────────────────────────────────────────────────────────────
   100→// Default weekday order is Sun..Sat. When the user has chosen Monday or
   101→// Saturday as their week start, we rotate the header array to match.
   102→// (BUGHUNT-OTHER-1 BUG-H3: previously hardcoded `weekStartsOn: 0` (Sunday)
   103→//  and ignored the user's `weekStart` setting entirely.)
   104→const WEEKDAYS_BASE = ['Min', 'Sen', 'Sel', 'Rab', 'Kam', 'Jum', 'Sab'];
   105→
   106→function weekStartsOnNum(weekStart?: string | null): 0 | 1 | 6 {
   107→  if (weekStart === 'saturday') return 6;
   108→  if (weekStart === 'sunday') return 0;
   109→  // Default to Monday (matches the Settings UI default).
   110→  return 1;
   111→}
   112→
   113→function rotateWeekdays(weekStartsOn: 0 | 1 | 6): string[] {
   114→  const idx = weekStartsOn === 6 ? 6 : weekStartsOn; // 0 or 1 or 6
   115→  return [...WEEKDAYS_BASE.slice(idx), ...WEEKDAYS_BASE.slice(0, idx)];
   116→}

   118→// PREMIUM REDESIGN (Rutina Aurora / Task 4-c): legacy gray/orange/lime
   119→// heatmap → Aurora teal-emerald scale. 0% keeps a soft destructive tint —
   120→// a tracked-but-missed day is a *status*, not a heat level. High cells
   121→// (bg-teal-600/500) need white text for AA contrast; see the text helpers.
   122→function getHeatmapColor(rate: number | null): string {
   123→  if (rate === null) return 'bg-muted/60';
   124→  if (rate === 0) return 'bg-destructive/25 dark:bg-destructive/15';
   125→  if (rate < 50) return 'bg-teal-200/70 dark:bg-teal-900/50';
   126→  if (rate < 75) return 'bg-teal-400/80 dark:bg-teal-700/60';
   127→  return 'bg-teal-600 dark:bg-teal-500';
   128→}
   129→
   130→function getHeatmapTextColor(rate: number | null): string {
   131→  if (rate === null) return 'text-muted-foreground';
   132→  if (rate === 0) return 'text-destructive dark:text-destructive/80';
   133→  if (rate >= 75) return 'text-white';
   134→  return 'text-foreground';
   135→}
   136→
   137→// Day-number color follows the same rules (white on high-heat cells so it
   138→// stays readable on teal-600; today keeps its primary-bold affordance).
   139→function getDayNumTextColor(day: DayData): string {
   140→  if (day.completionRate !== null && day.completionRate >= 75) return 'text-white font-bold';
   141→  if (day.isToday) return 'text-primary font-bold';
   142→  if (!day.isCurrentMonth) return 'text-muted-foreground';
   143→  return 'text-foreground';
   144→}
   145→
   146→// Task 4-c hover affordance: neutral/destructive cells tint with accent on
   147→// hover (spec: `hover:bg-accent/60`); heat-colored cells keep their fill —
   148→// the accent tint would wash out the white % text — so they brighten + get
   149→// a hairline ring instead. Both paths get active:scale-95 press feedback.
   150→function getHeatmapHover(rate: number | null): string {
   151→  if (rate === null || rate === 0) return 'hover:bg-accent/60';
   152→  return 'hover:brightness-105 hover:ring-1 hover:ring-ring/60';
   153→}

   166→// Task 4-c: Indonesian aria-label for the day buttons, e.g.
   167→// "Rabu 15 Januari 2025, 3 dari 5 habit selesai".
   168→function buildDayAriaLabel(day: DayData): string {
   169→  const dateLabel = `${format(day.date, 'EEEE', {
   170→    locale: idLocale,
   171→  })} ${format(day.date, 'd MMMM yyyy', { locale: idLocale })}`;
   172→  if (!day.isCurrentMonth) return dateLabel;
   173→  if (day.completionRate === null) return `${dateLabel}, belum ada data`;
   174→  if (day.totalHabits > 0) {
   175→    return `${dateLabel}, ${day.completedHabits} dari ${day.totalHabits} habit selesai`;
   176→  }
   177→  // Edge case: logs exist but no habit was considered active that day
   178→  // (completionRate is 100 by convention — see BUG-9 note above).
   179→  return `${dateLabel}, semua catatan selesai`;
   180→}

   182→function generateMonthOptions(): { value: string; label: string }[] {
   183→  const now = new Date();
   184→  const options: { value: string; label: string }[] = [];
   185→  // Show last 12 months and next 3 months
   186→  for (let i = -12; i <= 3; i++) {
   187→    const d = new Date(now.getFullYear(), now.getMonth() + i, 1);
   188→    options.push({
   189→      value: format(d, 'yyyy-MM'),
   190→      label: format(d, 'MMMM yyyy'),
   191→    });
   192→  }
   193→  return options;
   194→}

   196→// ── Component ──────────────────────────────────────────────────────────────
   197→export default function CalendarView() {
   198→  const selectedMonth = useAppStore(s => s.selectedMonth);
   199→  const setSelectedMonth = useAppStore(s => s.setSelectedMonth);
   200→  const queryClient = useQueryClient();
   201→
   202→  // ── Fetch AppSettings to read `weekStart` (BUG-H3 fix) ────────────────
   203→  // Shares the ['settings'] cache with the Settings page so changes apply
   204→  // immediately after a save + invalidation.
   205→  const { data: settings = null } = useQuery<AppSettings | null>({
   206→    queryKey: ['settings'],
   207→    queryFn: async () => {
   208→      const res = await fetch('/api/settings');
   209→      if (!res.ok) return null;
   210→      return res.json();
   211→    },
   212→    staleTime: 60_000,
   213→  });
   214→  const weekStartsOn = weekStartsOnNum(settings?.weekStart);
   215→  const WEEKDAYS = useMemo(() => rotateWeekdays(weekStartsOn), [weekStartsOn]);
   216→
   217→  const monthOptions = useMemo(() => generateMonthOptions(), []);
   218→
   219→  // ── Parse current month ────────────────────────────────────────────────
   220→  const monthDate = useMemo(() => {
   221→    const [y, m] = selectedMonth.split('-').map(Number);
   222→    return new Date(y, m - 1, 1);
   223→  }, [selectedMonth]);
   224→
   225→  const monthLabel = useMemo(
   226→    () => format(monthDate, 'MMMM yyyy'),
   227→    [monthDate]
   228→  );
   229→
   230→  // ── Fetch data (TanStack Query) ───────────────────────────────────────
   231→  const { data: habits = [], isLoading: habitsLoading } = useQuery<Habit[]>({
   232→    queryKey: ['habits'],
   233→    queryFn: async () => {
   234→      const res = await fetch('/api/habits');
   235→      if (!res.ok) return [];
   236→      return res.json();
   237→    },
   238→    staleTime: 30_000,
   239→  });
   240→
   241→  const { data: dailyLogs = [] } = useQuery<DailyLog[]>({
   242→    queryKey: ['daily-logs-month', selectedMonth],
   243→    queryFn: async () => {
   244→      const res = await fetch(`/api/daily-logs?month=${selectedMonth}`);
   245→      if (!res.ok) return [];
   246→      return res.json();
   247→    },
   248→    staleTime: 30_000,
   249→  });
   250→
   251→  const habitIds = habits.map(h => h.id).join(',');
   252→  const { data: habitLogs = [], isError: fetchError } = useQuery<HabitLog[]>({
   253→    queryKey: ['habit-logs-batch', selectedMonth, habitIds],
   254→    queryFn: async () => {
   255→      if (!habitIds) return [];
   256→      const res = await fetch(`/api/habits/batch-logs?month=${selectedMonth}&habitIds=${habitIds}`);
   257→      if (!res.ok) return [];
   258→      const groupedLogs: Record<string, HabitLog[]> = await res.json();
   259→      return Object.values(groupedLogs).flat();
   260→    },
   261→    enabled: habits.length > 0,
   262→    staleTime: 30_000,
   263→  });
   264→
   265→  const loading = habitsLoading;
   266→
   267→  // ── Build daily log lookup ─────────────────────────────────────────────
   268→  const dailyLogMap = useMemo(() => {
   269→    const map: Record<string, DailyLog> = {};
   270→    dailyLogs.forEach((d) => {
   271→      const key = d.date?.slice(0, 10);
   272→      if (key) map[key] = d;
   273→    });
   274→    return map;
   275→  }, [dailyLogs]);
   276→
   277→  // ── Build habit log lookup: dateStr -> { completed: number, total: number }
   278→  const habitLogMap = useMemo(() => {
   279→    const map: Record<
   280→      string,
   281→      { completed: number; total: number; habits: HabitLog[] }
   282→    > = {};
   283→    habitLogs.forEach((log) => {
   284→      const key = log.date?.slice(0, 10);
   285→      if (!key) return;
   286→      if (!map[key]) map[key] = { completed: 0, total: 0, habits: [] };
   287→      map[key].habits.push(log);
   288→      map[key].total++;
   289→      if (log.completed) map[key].completed++;
   290→    });
   291→    return map;
   292→  }, [habitLogs]);
   293→
   294→  // ── Build calendar days ────────────────────────────────────────────────
   295→  const calendarDays = useMemo<DayData[]>(() => {
   296→    const monthStart = startOfMonth(monthDate);
   297→    const monthEnd = endOfMonth(monthDate);
   298→    // BUGHUNT-OTHER-1 BUG-H3: respect user's `weekStart` setting instead
   299→    // of hardcoding Sunday.
   300→    const calStart = startOfWeek(monthStart, { weekStartsOn });
   301→    const calEnd = endOfWeek(monthEnd, { weekStartsOn });
   302→    const days = eachDayOfInterval({ start: calStart, end: calEnd });
   303→
   304→    const today = new Date();
   305→    // BUG-9 fix: count habits active on a SPECIFIC day (startDate <= day AND
   306→    // (endDate is null OR endDate >= day) AND status !== 'archived') instead
   307→    // of using the CURRENT total. Previously a user with 10 current habits
   308→    // looking at a day last month when they only had 3 habits would see
   309→    // completion rates divided by 10, drastically understating past days.
   310→    const activeHabitCountOnDay = (day: Date): number => {
   311→      const dayStart = startOfDay(day);
   312→      const ts = dayStart.getTime();
   313→      return habits.filter((h) => {
   314→        if (h.status === 'archived') return false;
   315→        // Parse startDate/endDate as local-midnight Date objects. They arrive
   316→        // as ISO strings from the API; Date-only strings ("yyyy-MM-dd") parse
   317→        // as UTC midnight, so we slice to 10 chars and use new Date(y,m,d) to
   318→        // avoid TZ-induced off-by-one.
   319→        if (h.startDate) {
   320→          const sd = h.startDate.slice(0, 10);
   321→          const [sy, sm, sd2] = sd.split('-').map(Number);
   322→          if (new Date(sy, sm - 1, sd2).getTime() > ts) return false;
   323→        }
   324→        if (h.endDate) {
   325→          const ed = h.endDate.slice(0, 10);
   326→          const [ey, em, ed2] = ed.split('-').map(Number);
   327→          if (new Date(ey, em - 1, ed2).getTime() < ts) return false;
   328→        }
   329→        return true;
   330→      }).length;
   331→    };
   332→
   333→    return days.map((d) => {
   334→      const dayStr = format(d, 'yyyy-MM-dd');
   335→      const hLogs = habitLogMap[dayStr];
   336→      const dLog = dailyLogMap[dayStr];
   337→      const inMonth = isSameMonth(d, monthDate);
   338→      const isFuture = isBefore(today, startOfDay(d)) && !isToday(d);
   339→
   340→      // BUG-9 + BUG-26 fix:
   341→      //  - Use per-day active habit count (not current total) → past days
   342→      //    aren't divided by an anachronistic habit count.
   343→      //  - No logs for the day → neutral (null), not 0% red. Showing 0% red
   344→      //    for a day before the user started tracking was misleading.
   345→      const totalHabitsOnDay = inMonth && !isFuture ? activeHabitCountOnDay(d) : 0;
   346→
   347→      let completionRate: number | null = null;
   348→      if (!inMonth) {
   349→        completionRate = null;
   350→      } else if (isFuture) {
   351→        completionRate = null;
   352→      } else if (hLogs && hLogs.total > 0 && totalHabitsOnDay > 0) {
   353→        // Has logs + habits were active that day: compute real rate.
   354→        completionRate = Math.round((hLogs.completed / totalHabitsOnDay) * 100);
   355→      } else if (hLogs && hLogs.total > 0 && totalHabitsOnDay === 0) {
   356→        // Edge case: logs exist but no habits are considered active that day
   357→        // (e.g. all habits were archived). Treat as 100% — the user did log
   358→        // something. Neutral (null) would also be defensible; 100% matches
   359→        // the spirit of "completed everything that was expected".
   360→        completionRate = 100;
   361→      } else {
   362→        // No logs at all → neutral, not 0% red (BUG-26).
   363→        completionRate = null;
   364→      }
   365→
   366→      return {
   367→        date: d,
   368→        dayStr,
   369→        dayNum: d.getDate(),
   370→        isCurrentMonth: inMonth,
   371→        isToday: isToday(d),
   372→        completionRate,
   373→        mood: dLog?.mood ?? null,
   374→        totalHabits: totalHabitsOnDay,
   375→        completedHabits: hLogs?.completed ?? 0,
   376→      };
   377→    });
   378→  }, [monthDate, habits, habitLogMap, dailyLogMap, weekStartsOn]);
   379→
   380→  // ── Month summary ──────────────────────────────────────────────────────
   381→  const monthSummary = useMemo(() => {
   382→    const activeDays = calendarDays.filter(
   383→      (d) =>
   384→        d.isCurrentMonth &&
   385→        d.completionRate !== null
   386→    );
   387→    if (activeDays.length === 0)
   388→      return { avg: 0, best: null, worst: null, entries: 0 };
   389→
   390→    const rates = activeDays.map((d) => (d.completionRate as number) || 0);
   391→    const avg = rates.reduce((a, b) => a + b, 0) / rates.length;
   392→    const bestIdx = rates.indexOf(Math.max(...rates));
   393→    const worstIdx = rates.indexOf(Math.min(...rates));
   394→
   395→    return {
   396→      avg,
   397→      best: activeDays[bestIdx],
   398→      worst: activeDays[worstIdx],
   399→      entries: activeDays.length,
   400→    };
   401→  }, [calendarDays]);
   402→
   403→  // ── Navigation handlers ────────────────────────────────────────────────
   404→  const goToPrevMonth = useCallback(() => {
   405→    const prev = subMonths(monthDate, 1);
   406→    setSelectedMonth(format(prev, 'yyyy-MM'));
   407→  }, [monthDate, setSelectedMonth]);
   408→
   409→  const goToNextMonth = useCallback(() => {
   410→    const next = addMonths(monthDate, 1);
   411→    setSelectedMonth(format(next, 'yyyy-MM'));
   412→  }, [monthDate, setSelectedMonth]);
   413→
   414→  // ── Render ─────────────────────────────────────────────────────────────
   415→  return (
   416→    <div className="space-y-6">
   417→      {/* ── Header with month navigation ──────────────────────────────── */}
   418→      <PageHeader
   419→        title="Kalender"
   420→        description="Visualisasikan penyelesaian habit kamu dalam heatmap bulanan."
   421→        action={
   422→          <div className="flex items-center gap-2">
   423→            <Button variant="outline" size="icon" onClick={goToPrevMonth}>
   424→              <ChevronLeft className="h-4 w-4" />
   425→            </Button>
   426→
   427→            <Select
   428→              value={selectedMonth}
   429→              onValueChange={setSelectedMonth}
   430→            >
   431→              <SelectTrigger className="w-[180px]">
   432→                <SelectValue />
   433→              </SelectTrigger>
   434→              <SelectContent>
   435→                {monthOptions.map((opt) => (
   436→                  <SelectItem key={opt.value} value={opt.value}>
   437→                    {opt.label}
   438→                  </SelectItem>
   439→                ))}
   440→              </SelectContent>
   441→            </Select>
   442→
   443→            <Button variant="outline" size="icon" onClick={goToNextMonth}>
   444→              <ChevronRight className="h-4 w-4" />
   445→            </Button>
   446→          </div>
   447→        }
   448→      />
   449→
   450→      {fetchError && (
   451→        <div className="flex flex-col items-center justify-center py-16 gap-3">
   452→          <p className="text-sm text-muted-foreground">Gagal memuat data kalender</p>
   453→          <Button variant="outline" size="sm" onClick={() => { queryClient.invalidateQueries({ queryKey: ['habits'] }); queryClient.invalidateQueries({ queryKey: ['daily-logs-month'] }); queryClient.invalidateQueries({ queryKey: ['habit-logs-batch'] }); }}>
   454→            <RefreshCw className="h-4 w-4" />
   455→            Coba Lagi
   456→          </Button>
   457→        </div>
   458→      )}
   459→
   460→      {!fetchError && (loading ? (
   461→        <Skeleton className="h-[600px] w-full rounded-xl" />
   462→      ) : (
   463→        <>
   464→          {/* ── Calendar Card ──────────────────────────────────────────── */}
   465→          <Card>
   466→            <CardContent className="p-4 md:p-6">
   467→              {/* Weekday headers */}
   468→              <div className="grid grid-cols-7 gap-1 mb-2">
   469→                {WEEKDAYS.map((d) => (
   470→                  <div
   471→                    key={d}
   472→                    className="text-center text-xs font-semibold text-muted-foreground py-2"
   473→                  >
   474→                    {d}
   475→                  </div>
   476→                ))}
   477→              </div>
   478→
   479→              {/* Day cells */}
   480→              <div className="grid grid-cols-7 gap-1">
   481→                {calendarDays.map((day, idx) => (
   482→                  <div
   483→                    key={idx}
   484→                    className={`
   485→                      relative min-h-[72px] sm:min-h-[88px] md:min-h-[100px] rounded-lg p-1.5 sm:p-2
   486→                      transition-all duration-150
   487→                      ${getHeatmapColor(day.completionRate)}
   488→                      ${!day.isCurrentMonth ? 'opacity-35' : ''}
   489→                      ${day.isToday ? 'ring-2 ring-primary ring-offset-2 ring-offset-background anim-glow-breathe' : ''}
   490→                      hover:ring-1 hover:ring-primary/50 cursor-default
   491→                    `}
   492→                  >
   493→                    <div className="flex items-center justify-between">
   494→                      <span
   495→                        className={`text-xs sm:text-sm font-medium ${
   496→                          day.isToday
   497→                            ? 'text-primary font-bold'
   498→                            : !day.isCurrentMonth
   499→                            ? 'text-muted-foreground'
   500→                            : 'text-foreground'
   501→                        }`}
   502→                      >
   503→                        {day.dayNum}
   504→                      </span>
   505→                    </div>
   506→
   507→                    {day.isCurrentMonth && day.completionRate !== null && (
   508→                      <div className="mt-1 flex flex-col items-center gap-0.5">
   509→                        <span
   510→                          className={`text-xs sm:text-xs font-bold ${getHeatmapTextColor(
   511→                            day.completionRate
   512→                          )}`}
   513→                        >
   514→                          {day.completionRate}%
   515→                        </span>
   516→                        <div className="w-full bg-black/10 rounded-full h-1 overflow-hidden">
   517→                          <div
   518→                            className="h-full rounded-full bg-primary transition-all duration-300"
   519→                            style={{
   520→                              width: `${Math.max(day.completionRate, 0)}%`,
   521→                            }}
   522→                          />
   523→                        </div>
   524→                      </div>
   525→                    )}
   526→
