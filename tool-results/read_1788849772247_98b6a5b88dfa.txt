   447→function PremiumBottomNav({
   448→  activeTab,
   449→  onNavClick,
   450→}: {
   451→  activeTab: TabId;
   452→  onNavClick: (id: TabId) => void;
   453→}) {
   454→  const dockRef = useRef<HTMLDivElement>(null);
   455→  const [dockW, setDockW] = useState(0);
   456→  const [fabOpen, setFabOpen] = useState(false);
   457→  // BUGHUNT-ROUND2 FAB-1: quick-add trigger lives in the store so this
   458→  // deep nav component can fire it without prop-drilling from Home().
   459→  const triggerQuickAdd = useAppStore((s) => s.triggerQuickAdd);
   460→
   461→  useLayoutEffect(() => {
   462→    if (!dockRef.current) return;
   463→    const update = () => setDockW(dockRef.current?.offsetWidth ?? 0);
   464→    update();
   465→    const observer = new ResizeObserver(update);
   466→    observer.observe(dockRef.current);
   467→    return () => observer.disconnect();
   468→  }, []);
   469→
   470→  // Close popup on Escape (a11y parity with the tap-to-close backdrop).
   471→  useEffect(() => {
   472→    if (!fabOpen) return;
   473→    const onKey = (e: KeyboardEvent) => {
   474→      if (e.key === 'Escape') setFabOpen(false);
   475→    };
   476→    window.addEventListener('keydown', onKey);
   477→    return () => window.removeEventListener('keydown', onKey);
   478→  }, [fabOpen]);
   479→
   480→  // Liquid indicator geometry. GPU-friendly: fixed width per tab +

   514→            )}
   515→            strokeWidth={isActive ? 2.4 : 1.7}
   516→          />
   517→        </span>
   518→        <span
   519→          className={cn(
   520→            'premium-dock-label transition-colors duration-300 motion-reduce:transition-none',
   521→            isActive
   522→              ? 'font-bold text-white'
   523→              : 'font-medium text-slate-500 dark:text-slate-400'

   526→          {item.label}

   532→  return (
   533→    <div className="fixed bottom-0 inset-x-0 z-40 md:hidden">

   556→        )}
   557→        style={{ height: DOCK_H }}
   558→      >
   559→        {/* Glass hairline highlight along the top edge (light refraction) */}
   560→        <div
   561→          aria-hidden="true"
   562→          className="absolute inset-x-[16px] top-0 h-px rounded-full bg-gradient-to-r from-transparent via-white/80 to-transparent dark:via-white/15 pointer-events-none"

   564→
   565→        {/* Liquid gradient indicator — slides between tabs, passes behind
   566→            the FAB (z-0 vs z-20). Fades out when a non-dock tab (goals)

   568→        <div
   569→          aria-hidden="true"
   570→          className="nav-liquid-indicator absolute left-0 top-[7px] z-0 rounded-full"
   571→          style={{
   572→            height: DOCK_H - 14,
   573→            width: Math.max(0, tabW - IND_INSET * 2),
   574→            transform: `translateX(${indX}px)`,

   608→            />

   633→              onClick={() => { triggerQuickAdd('income'); onNavClick('finance'); setFabOpen(false); }}
