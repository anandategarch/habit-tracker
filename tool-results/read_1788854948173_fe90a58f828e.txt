    99→// ── Helpers ────────────────────────────────────────────────────────────────
   100→// Default weekday order is Sun..Sat. When the user has chosen Monday or
   101→// Saturday as their week start, we rotate the header array to match.
   102→// (BUGHUNT-OTHER-1 BUG-H3: previously hardcoded `weekStartsOn: 0` (Sunday)
   103→//  and ignored the user's `weekStart` setting entirely.)
   104→const WEEKDAYS_BASE = ['Min', 'Sen', 'Sel', 'Rab', 'Kam', 'Jum', 'Sab'];
   105→
   106→function weekStartsOnNum(weekStart?: string | null): 0 | 1 | 6 {
   107→  if (weekStart === 'saturday') return 6;
   108→  if (weekStart === 'sunday') return 0;
   109→  // Default to Monday (matches the Settings UI default).
   110→  return 1;
   111→}
   112→
   113→function rotateWeekdays(weekStartsOn: 0 | 1 | 6): string[] {
   114→  const idx = weekStartsOn === 6 ? 6 : weekStartsOn; // 0 or 1 or 6
   115→  return [...WEEKDAYS_BASE.slice(idx), ...WEEKDAYS_BASE.slice(0, idx)];
   116→}

   118→// PREMIUM REDESIGN (Rutina Aurora / Task 4-c): legacy gray/orange/lime
   119→// heatmap → Aurora teal-emerald scale. 0% keeps a soft destructive tint —
   120→// a tracked-but-missed day is a *status*, not a heat level. High cells
   121→// (bg-teal-600/500) need white text for AA contrast; see the text helpers.
   122→function getHeatmapColor(rate: number | null): string {
   123→  if (rate === null) return 'bg-muted/60';
   124→  if (rate === 0) return 'bg-destructive/25 dark:bg-destructive/15';
   125→  if (rate < 50) return 'bg-teal-200/70 dark:bg-teal-900/50';
   126→  if (rate < 75) return 'bg-teal-400/80 dark:bg-teal-700/60';
   127→  return 'bg-teal-600 dark:bg-teal-500';
   128→}
   129→
   130→function getHeatmapTextColor(rate: number | null): string {
   131→  if (rate === null) return 'text-muted-foreground';
   132→  if (rate === 0) return 'text-destructive dark:text-destructive/80';
   133→  if (rate >= 75) return 'text-white';
   134→  return 'text-foreground';
   135→}
   136→
   137→// Day-number color follows the same rules (white on high-heat cells so it
   138→// stays readable on teal-600; today keeps its primary-bold affordance).
   139→function getDayNumTextColor(day: DayData): string {
   140→  if (day.completionRate !== null && day.completionRate >= 75) return 'text-white font-bold';
   141→  if (day.isToday) return 'text-primary font-bold';
   142→  if (!day.isCurrentMonth) return 'text-muted-foreground';
   143→  return 'text-foreground';
   144→}

   166→// Task 4-c: Indonesian aria-label for the day buttons, e.g.
   167→// "Rabu 15 Januari 2025, 3 dari 5 habit selesai".
   168→function buildDayAriaLabel(day: DayData): string {
   169→  const dateLabel = `${format(day.date, 'EEEE', {
   170→    locale: idLocale,
   171→  })} ${format(day.date, 'd MMMM yyyy', { locale: idLocale })}`;
   172→  if (!day.isCurrentMonth) return dateLabel;
   173→  if (day.completionRate === null) return `${dateLabel}, belum ada data`;
   174→  if (day.totalHabits > 0) {
   175→    return `${dateLabel}, ${day.completedHabits} dari ${day.totalHabits} habit selesai`;
   176→  }
   177→  // Edge case: logs exist but no habit was considered active that day
   178→  // (completionRate is 100 by convention — see BUG-9 note above).
   179→  return `${dateLabel}, semua catatan selesai`;
   180→}

   182→function generateMonthOptions(): { value: string; label: string }[] {
   183→  const now = new Date();
   184→  const options: { value: string; label: string }[] = [];
   185→  // Show last 12 months and next 3 months
   186→  for (let i = -12; i <= 3; i++) {
   187→    const d = new Date(now.getFullYear(), now.getMonth() + i, 1);
   188→    options.push({
   189→      value: format(d, 'yyyy-MM'),
   190→      label: format(d, 'MMMM yyyy'),
   191→    });
   192→  }
   193→  return options;
   194→}

   205→// ── Component ──────────────────────────────────────────────────────────────
   206→export default function CalendarView() {
   207→  const selectedMonth = useAppStore(s => s.selectedMonth);
   208→  const setSelectedMonth = useAppStore(s => s.setSelectedMonth);
   209→  // Task 4-c (calendar 1-click): store primitive from 4-foundation — jumps
   210→  // to the tracker grid with the tapped date preselected. dayStr is
   211→  // 'yyyy-MM-dd', exactly matching the store's selectedDate semantics.
   212→  const openTrackerDate = useAppStore(s => s.openTrackerDate);
   213→  const queryClient = useQueryClient();
   214→
   215→  // ── Fetch AppSettings to read `weekStart` (BUG-H3 fix) ────────────────
   216→  // Shares the ['settings'] cache with the Settings page so changes apply
   217→  // immediately after a save + invalidation.
   218→  const { data: settings = null } = useQuery<AppSettings | null>({
   219→    queryKey: ['settings'],
   220→    queryFn: async () => {
   221→      const res = await fetch('/api/settings');
   222→      if (!res.ok) return null;
   223→      return res.json();
   224→    },
   225→    staleTime: 60_000,
   226→  });
   227→  const weekStartsOn = weekStartsOnNum(settings?.weekStart);
   228→  const WEEKDAYS = useMemo(() => rotateWeekdays(weekStartsOn), [weekStartsOn]);
   229→
   230→  const monthOptions = useMemo(() => generateMonthOptions(), []);
   231→
   232→  // ── Parse current month ────────────────────────────────────────────────
   233→  const monthDate = useMemo(() => {
   234→    const [y, m] = selectedMonth.split('-').map(Number);
   235→    return new Date(y, m - 1, 1);
   236→  }, [selectedMonth]);
   237→
   238→  const monthLabel = useMemo(
   239→    () => format(monthDate, 'MMMM yyyy'),
   240→    [monthDate]
   241→  );
   242→
   243→  // ── Fetch data (TanStack Query) ───────────────────────────────────────
   244→  const { data: habits = [], isLoading: habitsLoading } = useQuery<Habit[]>({
   245→    queryKey: ['habits'],
   246→    queryFn: async () => {
   247→      const res = await fetch('/api/habits');
   248→      if (!res.ok) return [];
   249→      return res.json();
   250→    },
   251→    staleTime: 30_000,
   252→  });
   253→
   254→  const { data: dailyLogs = [] } = useQuery<DailyLog[]>({
   255→    queryKey: ['daily-logs-month', selectedMonth],
   256→    queryFn: async () => {
   257→      const res = await fetch(`/api/daily-logs?month=${selectedMonth}`);
   258→      if (!res.ok) return [];
   259→      return res.json();
   260→    },
   261→    staleTime: 30_000,
   262→  });
   263→
   264→  const habitIds = habits.map(h => h.id).join(',');
   265→  const { data: habitLogs = [], isError: fetchError } = useQuery<HabitLog[]>({
   266→    queryKey: ['habit-logs-batch', selectedMonth, habitIds],
   267→    queryFn: async () => {
   268→      if (!habitIds) return [];
   269→      const res = await fetch(`/api/habits/batch-logs?month=${selectedMonth}&habitIds=${habitIds}`);
   270→      if (!res.ok) return [];
   271→      const groupedLogs: Record<string, HabitLog[]> = await res.json();
   272→      return Object.values(groupedLogs).flat();
   273→    },
   274→    enabled: habits.length > 0,
   275→    staleTime: 30_000,
   276→  });
   277→
   278→  const loading = habitsLoading;
   279→
   280→  // ── Build daily log lookup ─────────────────────────────────────────────
   281→  const dailyLogMap = useMemo(() => {
   282→    const map: Record<string, DailyLog> = {};
   283→    dailyLogs.forEach((d) => {
   284→      const key = d.date?.slice(0, 10);
   285→      if (key) map[key] = d;
   286→    });
   287→    return map;
   288→  }, [dailyLogs]);
   289→
   290→  // ── Build habit log lookup: dateStr -> { completed: number, total: number }
   291→  const habitLogMap = useMemo(() => {
   292→    const map: Record<
   293→      string,
   294→      { completed: number; total: number; habits: HabitLog[] }
   295→    > = {};
   296→    habitLogs.forEach((log) => {
   297→      const key = log.date?.slice(0, 10);
   298→      if (!key) return;
   299→      if (!map[key]) map[key] = { completed: 0, total: 0, habits: [] };
   300→      map[key].habits.push(log);
   301→      map[key].total++;
   302→      if (log.completed) map[key].completed++;
   303→    });
   304→    return map;
   305→  }, [habitLogs]);
   306→
   307→  // ── Build calendar days ────────────────────────────────────────────────
   308→  const calendarDays = useMemo<DayData[]>(() => {
   309→    const monthStart = startOfMonth(monthDate);
   310→    const monthEnd = endOfMonth(monthDate);
   311→    // BUGHUNT-OTHER-1 BUG-H3: respect user's `weekStart` setting instead
   312→    // of hardcoding Sunday.
   313→    const calStart = startOfWeek(monthStart, { weekStartsOn });
   314→    const calEnd = endOfWeek(monthEnd, { weekStartsOn });
   315→    const days = eachDayOfInterval({ start: calStart, end: calEnd });
   316→
   317→    const today = new Date();
   318→    // BUG-9 fix: count habits active on a SPECIFIC day (startDate <= day AND
   319→    // (endDate is null OR endDate >= day) AND status !== 'archived') instead
   320→    // of using the CURRENT total. Previously a user with 10 current habits
   321→    // looking at a day last month when they only had 3 habits would see
   322→    // completion rates divided by 10, drastically understating past days.
   323→    const activeHabitCountOnDay = (day: Date): number => {
   324→      const dayStart = startOfDay(day);
   325→      const ts = dayStart.getTime();
   326→      return habits.filter((h) => {
   327→        if (h.status === 'archived') return false;
   328→        // Parse startDate/endDate as local-midnight Date objects. They arrive
   329→        // as ISO strings from the API; Date-only strings ("yyyy-MM-dd") parse
   330→        // as UTC midnight, so we slice to 10 chars and use new Date(y,m,d) to
   331→        // avoid TZ-induced off-by-one.
   332→        if (h.startDate) {
   333→          const sd = h.startDate.slice(0, 10);
   334→          const [sy, sm, sd2] = sd.split('-').map(Number);
   335→          if (new Date(sy, sm - 1, sd2).getTime() > ts) return false;
   336→        }
   337→        if (h.endDate) {
   338→          const ed = h.endDate.slice(0, 10);
   339→          const [ey, em, ed2] = ed.split('-').map(Number);
   340→          if (new Date(ey, em - 1, ed2).getTime() < ts) return false;
   341→        }
   342→        return true;
   343→      }).length;
   344→    };
   345→
   346→    return days.map((d) => {
   347→      const dayStr = format(d, 'yyyy-MM-dd');
   348→      const hLogs = habitLogMap[dayStr];
   349→      const dLog = dailyLogMap[dayStr];
   350→      const inMonth = isSameMonth(d, monthDate);
   351→      const isFuture = isBefore(today, startOfDay(d)) && !isToday(d);
   352→
   353→      // BUG-9 + BUG-26 fix:
   354→      //  - Use per-day active habit count (not current total) → past days
   355→      //    aren't divided by an anachronistic habit count.
   356→      //  - No logs for the day → neutral (null), not 0% red. Showing 0% red
   357→      //    for a day before the user started tracking was misleading.
   358→      const totalHabitsOnDay = inMonth && !isFuture ? activeHabitCountOnDay(d) : 0;
   359→
   360→      let completionRate: number | null = null;
   361→      if (!inMonth) {
   362→        completionRate = null;
   363→      } else if (isFuture) {
   364→        completionRate = null;
   365→      } else if (hLogs && hLogs.total > 0 && totalHabitsOnDay > 0) {
   366→        // Has logs + habits were active that day: compute real rate.
   367→        completionRate = Math.round((hLogs.completed / totalHabitsOnDay) * 100);
   368→      } else if (hLogs && hLogs.total > 0 && totalHabitsOnDay === 0) {
   369→        // Edge case: logs exist but no habits are considered active that day
   370→        // (e.g. all habits were archived). Treat as 100% — the user did log
   371→        // something. Neutral (null) would also be defensible; 100% matches
   372→        // the spirit of "completed everything that was expected".
   373→        completionRate = 100;
   374→      } else {
   375→        // No logs at all → neutral, not 0% red (BUG-26).
   376→        completionRate = null;
   377→      }
   378→
   379→      return {
   380→        date: d,
   381→        dayStr,
   382→        dayNum: d.getDate(),
   383→        isCurrentMonth: inMonth,
   384→        isToday: isToday(d),
   385→        completionRate,
   386→        mood: dLog?.mood ?? null,
   387→        totalHabits: totalHabitsOnDay,
   388→        completedHabits: hLogs?.completed ?? 0,
   389→      };
   390→    });
   391→  }, [monthDate, habits, habitLogMap, dailyLogMap, weekStartsOn]);
   392→
   393→  // ── Month summary ──────────────────────────────────────────────────────
   394→  const monthSummary = useMemo(() => {
   395→    const activeDays = calendarDays.filter(
   396→      (d) =>
   397→        d.isCurrentMonth &&
   398→        d.completionRate !== null
   399→    );
   400→    if (activeDays.length === 0)
   401→      return { avg: 0, best: null, worst: null, entries: 0 };
   402→
   403→    const rates = activeDays.map((d) => (d.completionRate as number) || 0);
   404→    const avg = rates.reduce((a, b) => a + b, 0) / rates.length;
   405→    const bestIdx = rates.indexOf(Math.max(...rates));
   406→    const worstIdx = rates.indexOf(Math.min(...rates));
   407→
   408→    return {
   409→      avg,
   410→      best: activeDays[bestIdx],
   411→      worst: activeDays[worstIdx],
   412→      entries: activeDays.length,
   413→    };
   414→  }, [calendarDays]);
   415→
   416→  // ── Navigation handlers ────────────────────────────────────────────────
   417→  const goToPrevMonth = useCallback(() => {
   418→    const prev = subMonths(monthDate, 1);
   419→    setSelectedMonth(format(prev, 'yyyy-MM'));
   420→  }, [monthDate, setSelectedMonth]);
   421→
   422→  const goToNextMonth = useCallback(() => {
   423→    const next = addMonths(monthDate, 1);
   424→    setSelectedMonth(format(next, 'yyyy-MM'));
   425→  }, [monthDate, setSelectedMonth]);
   426→
   427→  // Task 4-c: day-cell tap → openTrackerDate (tracker grid + preselected
   428→  // date). zustand setters are stable, so the callback identity is stable.
   429→  const handleDayClick = useCallback(
   430→    (dayStr: string) => openTrackerDate(dayStr),
   431→    [openTrackerDate]
   432→  );
   433→
   434→  const retryFetch = useCallback(() => {
   435→    queryClient.invalidateQueries({ queryKey: ['habits'] });
   436→    queryClient.invalidateQueries({ queryKey: ['daily-logs-month'] });
   437→    queryClient.invalidateQueries({ queryKey: ['habit-logs-batch'] });
   438→  }, [queryClient]);
   439→
   440→  // ── Render ─────────────────────────────────────────────────────────────
   441→  return (
   442→    <div className="space-y-6">
   443→      {/* ── Header with month navigation ──────────────────────────────── */}
   444→      <PageHeader
   445→        title="Kalender"
   446→        description="Visualisasikan penyelesaian habit kamu dalam heatmap bulanan."
   447→        action={
   448→          <div className="flex items-center gap-2">
   449→            <Button variant="outline" size="icon" onClick={goToPrevMonth}>
   450→              <ChevronLeft className="h-4 w-4" />
   451→            </Button>
   452→
   453→            <Select
   454→              value={selectedMonth}
   455→              onValueChange={setSelectedMonth}
   456→            >
   457→              <SelectTrigger className="w-[180px]">
   458→                <SelectValue />
   459→              </SelectTrigger>
   460→              <SelectContent>
   461→                {monthOptions.map((opt) => (
   462→                  <SelectItem key={opt.value} value={opt.value}>
   463→                    {opt.label}
   464→                  </SelectItem>
   465→                ))}
   466→              </SelectContent>
   467→            </Select>
   468→
   469→            <Button variant="outline" size="icon" onClick={goToNextMonth}>
   470→              <ChevronRight className="h-4 w-4" />
   471→            </Button>
   472→          </div>
   473→        }
   474→      />
   475→
   476→      {fetchError && (
   477→        <div className="premium-card premium-empty rounded-2xl">
   478→          <div className="premium-empty-orb" aria-hidden="true">
   479→            <CalendarDays className="h-8 w-8 text-primary" />
   480→          </div>
   481→          <p className="text-sm font-medium text-foreground">
   482→            Gagal memuat data kalender
   483→          </p>
   484→          <p className="text-xs text-muted-foreground/70 -mt-0.5">
   485→            Coba muat ulang habit dan log kamu.
   486→          </p>
   487→          <Button
   488→            size="sm"
   489→            className="btn-primary-gradient anim-press"
   490→            onClick={retryFetch}
   491→          >
   492→            <RefreshCw className="h-4 w-4" />
   493→            Coba Lagi
   494→          </Button>
   495→        </div>
   496→      )}
   497→
   498→      {!fetchError && (loading ? (
   499→        // Task 4-c: light skeleton that mimics the calendar grid (weekday
   500→        // header bars + a 7-col grid of rounded squares, capped at 420px)
   501→        // instead of one giant blank block.
   502→        <div
   503→          className="premium-card premium-card-sheen rounded-2xl p-4 sm:p-5"
   504→          aria-busy="true"
   505→          aria-label="Memuat kalender"
   506→        >
   507→          <div className="grid grid-cols-7 gap-1.5 mb-3">
   508→            {WEEKDAYS.map((d) => (
   509→              <Skeleton key={d} className="h-4 rounded-md" />
   510→            ))}
   511→          </div>
   512→          <div className="grid grid-cols-7 gap-1.5 max-h-[420px] overflow-hidden">
   513→            {Array.from({ length: 35 }).map((_, i) => (
   514→              <Skeleton
   515→                key={i}
   516→                className="h-[60px] sm:h-[68px] rounded-lg"
   517→                style={{ animationDelay: `${(i % 7) * 70}ms` }}
   518→              />
   519→            ))}
   520→          </div>
   521→        </div>
