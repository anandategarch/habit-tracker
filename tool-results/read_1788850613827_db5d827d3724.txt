    64→/** Live preview theme colors without saving to DB */
    65→function previewTheme(primary: string, secondary: string, theme: string) {
    66→  const isDark = theme === 'dark';
    67→  applyThemeMode(isDark);
    68→
    69→  if (
    70→    primary.toLowerCase() === '#22c55e' &&
    71→    secondary.toLowerCase() === '#10b981'
    72→  ) {
    73→    resetThemeColors();
    74→  } else {
    75→    applyThemeColors(primary, secondary, isDark);
    76→  }
    77→}
