   421→
   422→function FlutterBottomNav({
   423→  activeTab,
   424→  onNavClick,
   425→}: {
   426→  activeTab: TabId;
   427→  onNavClick: (id: TabId) => void;
   428→}) {
   429→  const navRef = useRef<HTMLDivElement>(null);
   430→  const [navWidth, setNavWidth] = useState(388);
   431→  const [fabOpen, setFabOpen] = useState(false);
   432→  // BUGHUNT-ROUND2 FAB-1: quick-add trigger lives in the store so this
   433→  // deep nav component can fire it without prop-drilling from Home().
   434→  const triggerQuickAdd = useAppStore((s) => s.triggerQuickAdd);
   435→
   436→  useLayoutEffect(() => {
   437→    if (!navRef.current) return;
   438→    const update = () => setNavWidth(navRef.current?.offsetWidth ?? 388);
   439→    update();
   440→    const observer = new ResizeObserver(update);
   441→    observer.observe(navRef.current);
   442→    return () => observer.disconnect();
   443→  }, []);
   444→
   445→  // Close FAB popup on outside tap
   446→  useEffect(() => {
   447→    if (!fabOpen) return;
   448→    const handler = () => setFabOpen(false);
   449→    const timer = setTimeout(() => document.addEventListener('click', handler), 0);
   450→    return () => {
   451→      clearTimeout(timer);
   452→      document.removeEventListener('click', handler);
   453→    };
   454→  }, [fabOpen]);
   455→
   456→  // Fixed center notch — always at navWidth/2
   457→  const W = navWidth;
   458→  const H = FLUTTER_NAV_HEIGHT;
   459→  const cR = FLUTTER_CORNER_R;
   460→  const nR = FLUTTER_NOTCH_R;
   461→  const nB = FLUTTER_NOTCH_BASE_W;
   462→  const nX = W / 2; // FIXED center
   463→  const bumpHeight = nR;
   464→
   465→  // BUGFIX FLUTTER-2 #2,#5,#6: Rewrite Bezier control points untuk smooth
   466→  // dome tanpa 90° corners + tanpa dip into nav body.
   467→  // C1 of left curve = (nX-nR+8, -bumpHeight*0.4) — co-linear dengan incoming
   468→  //   L direction (rightward + slightly up) → no 90° corner, no dip.
   469→  // C2 of left curve = (nX-12, -bumpHeight) — horizontal through peak → C1 continuity.
   470→  // Right curve = mirror.
   471→  const path = [
   472→    `M ${cR} 0`,
   473→    `L ${nX - nR} 0`,
   474→    `C ${nX - nR + 8} ${-bumpHeight * 0.4} ${nX - 12} ${-bumpHeight} ${nX} ${-bumpHeight}`,
   475→    `C ${nX + 12} ${-bumpHeight} ${nX + nR - 8} ${-bumpHeight * 0.4} ${nX + nR} 0`,
   476→    `L ${W - cR} 0`,
   477→    `A ${cR} ${cR} 0 0 1 ${W} ${cR}`,
   478→    `L ${W} ${H}`,
   479→    `L 0 ${H}`,
   480→    `L 0 ${cR}`,
   481→    `A ${cR} ${cR} 0 0 1 ${cR} 0`,
   482→    'Z',
   483→  ].join(' ');
   484→
