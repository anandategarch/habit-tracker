    56→// PERF-FIX (FIX-TIER3 / Fix 14): Format time from transaction date.
    57→// Hoisted to module scope — this function is stateless and was previously
    58→// re-created on every render. MUST use timeZone: 'Asia/Jakarta' explicitly
    59→// (without it, toLocaleTimeString uses the runtime's default TZ — UTC on
    60→// Vercel serverless, or the user's browser TZ locally). This caused the
    61→// time shown here to disagree with the time shown in the Daily Recap's
    62→// hourly heatmap + transactions list (which both use timeZone:
    63→// 'Asia/Jakarta'). Same code path as daily-recap.tsx's formatTxTime() —
    64→// keep them in sync.
    65→function formatTime(dateStr: string) {
    66→  try {
    67→    const d = new Date(dateStr);
    68→    return d.toLocaleTimeString('id-ID', {
    69→      timeZone: 'Asia/Jakarta',
    70→      hour: '2-digit',
    71→      minute: '2-digit',
    72→    });
    73→  } catch {
    74→    return '';
    75→  }
    76→}
    77→
    78→// PERF-FIX (FIX-TIER3 / Fix 14): Flat row representation for the virtualized
    79→// list. The original component rendered a nested structure (groups → txs),
    80→// which forced the browser to allocate a DOM node for every transaction
    81→// even when off-screen. For months with 100+ transactions this meant
    82→// hundreds of nodes living in the layout. Flattening groups into a single
    83→// row list lets @tanstack/react-virtual only render the visible window.
    84→type FlatRow =
    85→  | { kind: 'header'; group: GroupedTransaction }
    86→  | { kind: 'tx'; tx: Transaction; txIdx: number };
    87→
    88→// PERF-FIX: estimateSize callbacks must be stable (not re-created each
    89→// render) — the virtualizer uses referential equality to decide whether
    90→// to re-measure. Hoisting to module scope also avoids a fresh closure
    91→// per render, which would otherwise cause the virtualizer to reset its
    92→// measurements on every render of this component.
    93→function estimateRowSize(row: FlatRow | undefined): number {
    94→  if (!row) return 80;
    95→  // Header rows are short (single-line pill). Transaction rows are taller
    96→  // (icon + 2-line content + amount + action buttons).
    97→  // FIX: updated from 92 → 122 to match actual card height after spacing fix
    98→  // (16px padding × 2 + 78px content + 12px margin-bottom = 122px).
    99→  // Also enabled measureElement below for self-healing dynamic measurement.
   100→  return row.kind === 'header' ? 36 : 122;
   101→}

   404→                  key={vItem.key}

   408→                    position: 'absolute',
   409→                    top: 0,
   410→                    left: 0,
   411→                    width: '100%',
   412→                    transform: `translateY(${vItem.start}px)`,
   413→                  }}

   421→                      {row.group.totalIncome > 0 && (

   424→                    </div>
   425→                  ) : (
   426→                    <TransactionRow
   427→                      tx={row.tx}
   428→                      txIdx={row.txIdx}

   430→                      selectedTxIds={selectedTxIds}
   431→                      expandedTxId={expandedTxId}
   432→                      onToggleExpand={(id: string) =>
   433→                        setExpandedTxId(prev => (prev === id ? null : id))
   434→                      }
   435→                      getCategoryMeta={getCategoryMeta}
   436→                      getSourceEmoji={getSourceEmoji}
   437→                      onToggleSelectTx={onToggleSelectTx}
   438→                      onEditTx={onEditTx}
   439→                      onDeleteTx={onDeleteTx}
   440→                    />
   441→                  )}
   442→                </div>
   443→              );
   444→            })}
   445→          </div>
   446→        </div>
   447→      )}
   448→
   449→      {/* ── Total Expense Footer ──────────────────────────────── */}
   450→      {filteredTransactions.length > 0 && (
   451→        <div className="tx-total-footer flex items-center justify-between">
   452→          {/* FIN-BUG-7 fix: only show today-expense block when viewing the
   453→              current month. For other months, today's expense is always 0
   454→              (filteredTransactions is scoped to selectedMonth) — showing
   455→              "Rp 0" was misleading. */}
   456→          {isCurrentMonth ? (
   457→            <div>
   458→              <p className="text-xs text-muted-foreground">Total Pengeluaran Hari Ini</p>

   460→            </div>

   462→            <div />

   464→          <div className="text-right">
   465→            <p className="text-xs text-muted-foreground">Total Transaksi</p>

   467→          </div>
   468→        </div>
   469→      )}
   470→    </div>
   471→  );
   472→}

   474→// PERF-FIX (Fix 14): Extracted as a separate component so it can be
   475→// memoized in the future (e.g. React.memo) and so the parent's render
   476→// doesn't force re-creation of every visible row's JSX tree. Keeping

   479→interface TransactionRowProps {
   480→  tx: Transaction;
   481→  txIdx: number;
   482→  multiSelect: boolean;
   483→  selectedTxIds: Set<string>;
   484→  // SHADCN-PHASE-2: expand support. expandedTxId is the currently-expanded
   485→  // tx id (owned by parent). onToggleExpand flips it (id -> id, same id -> null).
   486→  expandedTxId: string | null;
   487→  onToggleExpand: (id: string) => void;
   488→  getCategoryMeta: (cat: string) => { emoji: string; color: string };
   489→  getSourceEmoji: (name: string) => string;
   490→  onToggleSelectTx: (id: string) => void;
   491→  onEditTx: (tx: Transaction) => void;
   492→  onDeleteTx: (id: string) => void;
   493→}

   495→function TransactionRow({
   496→  tx,
   497→  txIdx,
   498→  multiSelect,
   499→  selectedTxIds,
   500→  expandedTxId,
   501→  onToggleExpand,
   502→  getCategoryMeta,
   503→  getSourceEmoji,
   504→  onToggleSelectTx,
   505→  onEditTx,
   506→  onDeleteTx,
   507→}: TransactionRowProps) {
   508→  const meta = getCategoryMeta(tx.category);
   509→  const isExpense = tx.type === 'expense';
   510→  // SHADCN-PHASE-2: tap-to-expand state. parseTags is null-safe (returns []
   511→  // for missing/invalid tags), so legacy rows without the column render no
   512→  // expand section. hasExpandContent gates both the onClick handler and the
   513→  // expand panel render — cards with no notes AND no tags are not
   514→  // tappable (the edit button remains the only interaction), avoiding a
   515→  // confusing "tapped but nothing happened" UX.
   516→  const isExpanded = expandedTxId === tx.id;
   517→  const tags = parseTags(tx.tags);
   518→  const hasExpandContent = !!(tx.notes || tags.length > 0);
   519→  return (
   520→    <div className="relative anim-stagger" style={{ animationDelay: `${Math.min(txIdx, 8) * 30}ms` }}>
   521→      {/* Timeline node */}
   522→      <div
   523→        className="tx-node"
   524→        style={{
   525→          backgroundColor: isExpense ? '#ef4444' : '#22c55e',
   526→        }}
   527→      />
   528→
   529→      {/* Transaction card.
   530→          SHADCN-PHASE-2: added hover lift + tap-expand.
   531→          - transition-all duration-200: smooth border/box-shadow/transform.
   532→          - border border-transparent: reserves 1px so hover:border-primary/30
   533→            doesn't cause layout shift (the existing .tx-card CSS uses
   534→            box-shadow for its visual edge, not border, so without this the
   535→            card would grow by 1px on hover).
   536→          - motion-safe:hover:-translate-y-0.5 + motion-safe:hover:shadow-md:
   537→            the lift effect. Wrapped in motion-safe: so prefers-reduced-motion
   538→            users get NO lift (the constraint). Border-color change is NOT
   539→            motion, so it stays for everyone.
   540→          - active:scale-[0.98]: press feedback (also exists in .tx-card:active
   541→            CSS — redundant but harmless; explicit here for clarity).
   542→          onClick behavior:
   543→          - multiSelect mode: toggle selection (preserved from before).
   544→          - single-select + hasExpandContent: toggle expand.
   545→          - single-select + no expand content: no-op (use the edit button). */}
   546→      <div
   547→        className={cn(
   548→          'tx-card group',
   549→          'transition-all duration-200',
   550→          'border border-transparent hover:border-primary/30',
   551→          'motion-safe:hover:-translate-y-0.5 motion-safe:hover:shadow-md',
   552→          'active:scale-[0.98]'
   553→        )}
   554→        onClick={() => {
   555→          if (multiSelect) {
   556→            onToggleSelectTx(tx.id);
   557→          } else if (hasExpandContent) {
   558→            onToggleExpand(tx.id);
   559→          } else {
   560→            // BUGFIX POST-3 #4: Fallback ke edit dialog saat card tidak punya
   561→            // notes/tags untuk expand. Sebelum Phase 2, tap card = edit. Sekarang
   562→            // tap card = expand (jika ada content) atau edit (jika tidak ada).
   563→            onEditTx(tx);
   564→          }
   565→        }}
   566→      >
   567→        {/* Multi-select checkbox
   568→            BUG-FINANCE-CAL BUG-1: wrap the checkbox in a stopPropagation
   569→            container. Without this, clicking the checkbox bubbles up to the
   570→            tx-card's onClick (which toggles selection in multi-select mode),
   571→            so onCheckedChange AND the parent onClick BOTH fire — net-zero
   572→            selection change. The checkbox appeared broken: clicking it did
   573→            nothing. Now the wrapper swallows the click so only the checkbox's
   574→            onCheckedChange runs. */}
   575→        {multiSelect && (
   576→          <div
   577→            className="absolute top-2 right-2 z-10"
   578→            onClick={(e) => e.stopPropagation()}
   579→          >
   580→            <Checkbox
   581→              checked={selectedTxIds.has(tx.id)}
   582→              onCheckedChange={() => onToggleSelectTx(tx.id)}
   583→            />
   584→          </div>
   585→        )}
   586→
   587→        <div className="flex items-start gap-4">
   588→          {/* Category icon with colored circle background.
   589→              SHADCN-PHASE-2: switched from type-based red/green tint to the
   590→              category's own brand color (meta.color) at 12.5% opacity (hex
   591→              alpha "20" = 32/255). More meaningful — each category gets a
   592→              subtle tinted circle that matches its emoji color, instead of
   593→              every expense being red and every income being green.
   594→              meta.color is always a 6-digit hex from getCategoryMeta (falls
   595→              back to '#78716c' stone-gray), so `${meta.color}20` produces a
   596→              valid 8-digit hex (RRGGBBAA) — same pattern the previous
   597→              type-based tint used ('#ef444415'). */}
   598→          <div
   599→            className="w-9 h-9 rounded-full flex items-center justify-center text-base shrink-0"
   600→            style={{ backgroundColor: `${meta.color}20` }}
   601→          >
   602→            {meta.emoji}
   603→          </div>
   604→
   605→          {/* Content */}
   606→          <div className="flex-1 min-w-0">
   607→            {/* Top row: title + time */}
   608→            <div className="flex items-start justify-between gap-3">
   609→              <span className="text-sm font-semibold truncate flex items-center gap-1.5">
   610→                {tx.category}
   611→                {tx.groupId && (
   612→                  <span className="text-[9px] px-1 py-0 rounded-full bg-success/10 text-success dark:bg-success/15 dark:text-success/80 font-medium shrink-0">
   613→                    Split
   614→                  </span>
   615→                )}
   616→              </span>
   617→              <span className="text-xs text-muted-foreground shrink-0">
   618→                {formatTime(tx.date)}
   619→              </span>
   620→            </div>
   621→
   622→            {/* Bottom row: description + source + amount */}
   623→            <div className="flex items-end justify-between gap-3 mt-1.5">
   624→              <div className="min-w-0 flex-1">
   625→                {tx.description && (
   626→                  <p className="text-xs text-muted-foreground truncate">
   627→                    {tx.description}
   628→                  </p>
   629→                )}
   630→                <p className="text-xs text-muted-foreground flex items-center gap-1 mt-1">
   631→                  <span>{getSourceEmoji(tx.source || 'Kas')}</span>
   632→                  <span className="truncate">{tx.source || 'Kas'}</span>
   633→                </p>
   634→                {/* SHADCN-PHASE-2: tag badges moved into the tap-to-expand
   635→                    panel below (previously rendered inline here). Showing
   636→                    them only on expand declutters the default card view —
   637→                    the category emoji circle + amount + source are enough
   638→                    at-a-glance info; notes + tags are detail-level. */}
   639→              </div>
   640→
   641→              {/* Amount */}
   642→              <div className="shrink-0 text-right">
   643→                <span
   644→                  className={cn(
   645→                    'text-sm font-bold',
   646→                    isExpense ? 'text-destructive' : 'text-primary'
   647→                  )}
   648→                >
   649→                  {isExpense ? '-' : '+'}{formatRupiah(tx.amount)}
   650→                </span>
   651→              </div>
   652→            </div>
   653→          </div>
   654→
   655→          {/* Action buttons (visible on hover, always visible on mobile) */}
   656→          {!multiSelect && (
   657→            <div className="flex flex-col gap-1.5 sm:opacity-0 sm:group-hover:opacity-100 transition-opacity shrink-0 ml-1">
   658→              {/* Disable edit + delete for transfer transactions —
   659→                  they're linked pairs that can't be modified
   660→                  independently without corrupting balances.
   661→                  BUGHUNT-ROUND2 TRANSFER-DEL: delete now WORKS for
   662→                  transfers (the API atomically removes the sibling pair +
   663→                  reverts both balances), but edit stays disabled. */}
   664→              <Button
   665→                variant="ghost"
   666→                size="icon"
   667→                className="h-9 w-9"
   668→                onClick={(e) => { e.stopPropagation(); onEditTx(tx); }}
   669→                disabled={tx.category === 'Transfer Antar Sumber'}
   670→                title={tx.category === 'Transfer Antar Sumber' ? 'Transfer tidak bisa diedit' : 'Edit'}
   671→              >
   672→                <Edit3 className="h-3 w-3" />
   673→              </Button>
   674→              <Button
   675→                variant="ghost"
   676→                size="icon"
   677→                className="h-9 w-9 text-destructive hover:text-destructive"
   678→                onClick={(e) => {
   679→                  e.stopPropagation();
   680→                  // BUGHUNT-ROUND2 TRANSFER-DEL: transfers CAN be deleted now —
   681→                  // the API deletes the linked pair atomically.
   682→                  onDeleteTx(tx.id);
   683→                }}
   684→                title={tx.category === 'Transfer Antar Sumber' ? 'Hapus transfer (kedua sisi otomatis)' : 'Hapus'}
   685→              >
   686→                <Trash2 className="h-3 w-3" />
   687→              </Button>
   688→            </div>
   689→          )}
   690→        </div>
   691→
   692→        {/* SHADCN-PHASE-2: tap-to-expand panel (notes + tags).
   693→            Renders only when this card is expanded AND has content.
   694→            anim-tab-enter (defined in globals.css) provides a smooth
   695→            fade + slide-in from the right, and respects
   696→            prefers-reduced-motion (globals.css sets it to `none` under
   697→            the reduced-motion media query).
   698→            Virtualization note: the parent virtualizer uses
   699→            measureElement (ResizeObserver) on each row, so when this
   700→            panel mounts/unmounts and the row's height changes, the
   701→            virtualizer auto-re-measures — no manual invalidate needed.
   702→            This is why state lives in the parent: a row can unmount when
   703→            scrolled out of view and remount when scrolled back, and its
   704→            expand state survives because it's keyed by tx.id in the
   705→            parent, not held in row-local state. */}
   706→        {isExpanded && hasExpandContent && (
   707→          <div className="mt-2 pt-2 border-t border-border/50 text-xs text-muted-foreground anim-tab-enter">
   708→            {tx.notes && (
   709→              <p className="mb-1 break-words">{tx.notes}</p>
   710→            )}
   711→            {tags.length > 0 && (
   712→              <div className="flex gap-1 flex-wrap">
   713→                {tags.map((tag, i) => (
   714→                  <span
   715→                    key={`${tag}-${i}`}
   716→                    className="px-2 py-0.5 rounded-full bg-muted text-[10px] text-muted-foreground border border-border/60"
   717→                  >
   718→                    #{tag}
   719→                  </span>
   720→                ))}
   721→              </div>
   722→            )}
   723→          </div>
   724→        )}
   725→      </div>
   726→    </div>
   727→  );
   728→}
